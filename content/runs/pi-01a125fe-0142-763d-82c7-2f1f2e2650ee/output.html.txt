      art.style.setProperty("--tilt-x", (-x * 8).toFixed(1) + "px");
      art.style.setProperty("--tilt-y", (-y * 6).toFixed(1) + "px");
    });
    scene.addEventListener("pointerleave", () => {
      art.style.setProperty("--tilt-x", "0px");
      art.style.setProperty("--tilt-y", "0px");
    });

    const button = document.getElementById("pause");
    const label = document.getElementById("pause-label");
    button.addEventListener("click", () => {
      const paused = document.body.classList.toggle("paused");
      label.textContent = paused ? "Resume ride" : "Pause ride";
      button.setAttribute("aria-label", paused ? "Resume animation" : "Pause animation");
      button.querySelector(".pause-icon").textContent = paused ? "▶" : "Ⅱ";
    });
  })();
</script>
