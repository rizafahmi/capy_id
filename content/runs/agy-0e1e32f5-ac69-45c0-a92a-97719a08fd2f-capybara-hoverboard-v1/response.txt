  function drawMountainLayer(range, horizonY, speed, fill, rimColor) {
    range.offset = (range.offset + speed) % range.totalWidth;
    const pts = range.pts;
    const step = range.totalWidth / (pts.length - 1);

    ctx.save();
    ctx.fillStyle = fill;
    ctx.strokeStyle = rimColor;
    ctx.lineWidth = 1.6;

    for (let pass = 0; pass < 2; pass++) {
      const baseX = -range.offset + pass * range.totalWidth;
      if (baseX > width || baseX + range.totalWidth < 0) continue;

      ctx.beginPath();
      ctx.moveTo(baseX, horizonY);
      for (let i = 0; i < pts.length; i++) {
        ctx.lineTo(baseX + i * step, horizonY - pts[i].h);
      }
      ctx.lineTo(baseX + range.totalWidth, horizonY);
      ctx.closePath();
      ctx.fill();

      ctx.beginPath();
      for (let i = 0; i < pts.length; i++) {
        const px = baseX + i * step;
        const py = horizonY - pts[i].h;
        if (i === 0) ctx.moveTo(px, py);
        else ctx.lineTo(px, py);
      }
      ctx.stroke();
    }
    ctx.restore();
  }

        ctx.save();
        ctx.translate(-16, -62);
        ctx.rotate(a);
        ctx.beginPath();
        ctx.moveTo(0, 0);
        ctx.quadraticCurveTo(18, -10, 36, 4);
        ctx.quadraticCurveTo(18, 0, 0, 0);
        ctx.fill();
        ctx.restore();
      }
    } else {
      // Glowing Cyber Obelisk Crystal
      ctx.fillStyle = '#14082b';
      ctx.strokeStyle = '#ff2a85';
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.moveTo(-10, 0);
      ctx.lineTo(-4, -38);
      ctx.lineTo(2, -48);
      ctx.lineTo(8, -34);
      ctx.lineTo(12, 0);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
    }
    ctx.restore();
  }

  // Update Foreground Character & SVG Sub-parts
  function updateCharacter(dt) {
    // Smooth speed interpolation
    state.speedMultiplier += (state.targetSpeedMultiplier - state.speedMultiplier) * 0.08;

    // Update HUD Speedometer
    const targetKmh = Math.round(42 * state.speedMultiplier + Math.sin(state.time * 4) * 2 + (state.isJumping ? 14 : 0));
    state.currentSpeedKmh += (targetKmh - state.currentSpeedKmh) * 0.15;
    speedValueEl.textContent = Math.round(state.currentSpeedKmh);

    // Smooth steering toward pointer
    state.smoothX += (state.pointerX - state.smoothX) * 0.075;
    state.smoothY += (state.pointerY - state.smoothY) * 0.075;

    // Jump & 360 Flip Physics
    if (state.isJumping) {
      state.jumpY += state.jumpVy;
      state.jumpVy += 0.68; // gravity

      if (state.doingFlip) {
        state.flipAngle += 14.2;
        if (state.flipAngle > 360) state.flipAngle = 360;
      }

      if (state.jumpY >= 0) {
        state.jumpY = 0;
        state.jumpVy = 0;
        state.isJumping = false;
        state.doingFlip = false;
        state.flipAngle = 0;
        // Landing bounce on yuzu and knees
        state.yuzuVy = 4.5;
        statusLabel.textContent = state.turbo
          ? 'MAXIMUM OVERDRIVE // SUPERSONIC CAPY'
          : 'CHILL CRUISING // YUZU BALANCED';
      }
    }

    // Hover oscillation
    const hoverFreq = state.turbo ? 6.5 : 3.6;
    const hoverBob = Math.sin(state.time * hoverFreq) * (state.turbo ? 7 : 11);
    const boardPitch = state.smoothX * 11 + (state.turbo ? 4.5 : 1.5) + Math.cos(state.time * hoverFreq) * 2.2 + (state.isJumping && !state.doingFlip ? state.jumpVy * 1.1 : 0);

    // Position the main wrapper
    const translateX = state.smoothX * 65;
    const translateY = state.smoothY * 35 + hoverBob + state.jumpY;
    riderWrapper.style.transform = `translate3d(${translateX.toFixed(1)}px, ${translateY.toFixed(1)}px, 0)`;

    // Rotate master rig around board center (255, 340)
    const totalAngle = boardPitch - state.flipAngle;
    masterRig.setAttribute('transform', `rotate(${totalAngle.toFixed(2)} 255 320)`);

    // Subtle secondary suspension bob on Capybara torso/knees
    const kneeCompression = Math.sin(state.time * hoverFreq - 0.5) * 3.5 + (state.jumpY < 0 ? -3 : 0);
    capyGroup.setAttribute('transform', `translate(0, ${kneeCompression.toFixed(2)})`);

    // Head bob & serene groove
    const headPitch = Math.sin(state.time * hoverFreq - 0.9) * 2.2 + (state.turbo ? 2.5 : 0);
    const headBobY = Math.cos(state.time * hoverFreq) * 1.8;
    capyHeadGroup.setAttribute('transform', `translate(0, ${headBobY.toFixed(2)}) rotate(${headPitch.toFixed(2)} 310 190)`);

    // Occasional Cute Ear Flick
    if (state.earTwitchTime > 0) {
      state.earTwitchTime -= dt;
      state.earAngle = Math.sin(state.time * 48) * 18;
    } else {
      if (Math.random() < 0.008) {
        state.earTwitchTime = 0.28;
      }
      state.earAngle *= 0.85;
    }
    earNear.setAttribute('transform', `translate(274, 146) rotate(${state.earAngle.toFixed(1)})`);

    // Cyber Goggles slide over eye in Turbo mode
    const targetGoggles = state.turbo ? 1 : 0;
    state.gogglesProgress += (targetGoggles - state.gogglesProgress) * 0.14;
    const gogX = -state.gogglesProgress * 9;
    const gogY = state.gogglesProgress * 22;
    const gogRot = state.gogglesProgress * 8;
    cyberGoggles.setAttribute('transform', `translate(${gogX.toFixed(2)}, ${gogY.toFixed(2)}) rotate(${gogRot.toFixed(2)} 335 150)`);

    // Yuzu Fruit Physics (bounces gently on head while staying miraculously balanced)
    const yuzuSpring = -0.28 * state.yuzuY;
    state.yuzuVy = (state.yuzuVy + yuzuSpring) * 0.84;
    state.yuzuY += state.yuzuVy;
    const yuzuIdleBob = Math.sin(state.time * hoverFreq - 1.2) * 2.2;
    const yuzuTotalY = Math.min(3, state.yuzuY + yuzuIdleBob);
    const yuzuTilt = Math.sin(state.time * 4.2) * 5 - boardPitch * 0.35;
    yuzuGroup.setAttribute('transform', `translate(302, ${(121 + yuzuTotalY).toFixed(2)}) rotate(${yuzuTilt.toFixed(2)})`);

    // Dynamic Wind-Rippling Scarf Waves
    const scarfSpeed = state.turbo ? 18 : 10;
    const w1 = Math.sin(state.time * scarfSpeed) * 12;
    const w2 = Math.cos(state.time * scarfSpeed * 0.85) * 15;
    const w3 = Math.sin(state.time * scarfSpeed * 1.15 + 1) * 10;

    scarfTailMain.setAttribute(
      'd',
      `M 4,4 C -42,${(2 + w1 * 0.5).toFixed(1)} -88,${(22 + w2).toFixed(1)} -138,${(8 + w1).toFixed(1)} C -118,${(28 + w2).toFixed(1)} -95,${(38 + w3).toFixed(1)} -128,${(48 + w1).toFixed(1)} C -78,${(50 + w2 * 0.6).toFixed(1)} -38,${(32 + w3 * 0.4).toFixed(1)} 6,24 Z`
    );
    scarfTailBack.setAttribute(
      'd',
      `M 0,0 C -45,${(-8 - w2 * 0.5).toFixed(1)} -85,${(10 + w1).toFixed(1)} -124,${(-4 + w3).toFixed(1)} C -108,${(16 + w1).toFixed(1)} -80,${(26 + w2).toFixed(1)} -115,${(34 + w3).toFixed(1)} C -72,${(38 + w1 * 0.5).toFixed(1)} -35,22 0,18 Z`
    );
    scarfStripe.setAttribute(
      'd',
      `M -18,12 Q -68,${(24 + w2 * 0.8).toFixed(1)} -114,${(20 + w1 * 0.8).toFixed(1)}`
    );

    // Thruster Flame & Aft Jet Flicker
    const flameStretch = (state.turbo ? 1.45 : 1.0) + Math.sin(state.time * 38) * 0.18;
    const coneH = (56 * flameStretch).toFixed(1);
    rearFlame.setAttribute('points', `-28,0 28,0 44,${coneH} -44,${coneH}`);
    frontFlame.setAttribute('points', `-28,0 28,0 44,${coneH} -44,${coneH}`);

    const jetScaleX = (state.turbo ? 1.65 : 1.0) + Math.sin(state.time * 45) * 0.22;
    const jetScaleY = (state.turbo ? 1.25 : 1.0) + Math.cos(state.time * 40) * 0.12;
    jetPlumeOuter.setAttribute('transform', `scale(${jetScaleX.toFixed(2)}, ${jetScaleY.toFixed(2)})`);
    jetPlumeCore.setAttribute('transform', `scale(${(jetScaleX * 1.05).toFixed(2)}, ${jetScaleY.toFixed(2)})`);

    // Pulse anti-grav rings
    const pulse1 = (state.time * 3) % 1;
    const pulse2 = (state.time * 3 + 0.5) % 1;
    gravRing1.setAttribute('rx', (42 + pulse1 * 42).toFixed(1));
    gravRing1.setAttribute('ry', (6 + pulse1 * 8).toFixed(1));
    gravRing1.setAttribute('stroke-opacity', (0.85 * (1 - pulse1)).toFixed(2));

    gravRing2.setAttribute('rx', (42 + pulse2 * 42).toFixed(1));
    gravRing2.setAttribute('ry', (6 + pulse2 * 8).toFixed(1));
    gravRing2.setAttribute('stroke-opacity', (0.75 * (1 - pulse2)).toFixed(2));

    // Spawn canvas exhaust particles aligned with board position
    const boardWorldX = width * 0.5 + translateX;
    const boardWorldY = height * 0.5 + translateY;
    spawnExhaustParticles(boardWorldX, boardWorldY, state.turbo);
  }

  // Main Animation Loop
  let lastTime = performance.now();
  function animate(now) {
    const dt = Math.min((now - lastTime) / 1000, 0.05);
    lastTime = now;
    state.time += dt;

    updateCharacter(dt);
    drawBackground(dt);

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
})();
