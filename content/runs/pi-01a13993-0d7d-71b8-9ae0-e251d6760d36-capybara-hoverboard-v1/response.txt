<script>
  (() => {
    const speed = document.getElementById('speed');
    const value = document.getElementById('speedValue');
    const scene = document.getElementById('scene');
    const mood = document.getElementById('mood');
    const boost = document.getElementById('boost');
    let boosting = false;

    function updateSpeed() {
      value.textContent = speed.value.padStart(2, '0');
      if (!boosting) {
        scene.style.setProperty('--pace', Number(speed.value) / 8);
        mood.textContent = Number(speed.value) > 14 ? 'Still absolutely unbothered' : '100% unbothered';
      }
    }

    speed.addEventListener('input', updateSpeed);
    boost.addEventListener('click', () => {
      if (boosting) return;
      boosting = true;
      scene.classList.add('boosting');
      scene.style.setProperty('--pace', '3.8');
      mood.textContent = 'Wheee. But, like, calmly.';
      boost.setAttribute('aria-disabled', 'true');
      setTimeout(() => {
        scene.classList.remove('boosting');
        boost.removeAttribute('aria-disabled');
        boosting = false;
        updateSpeed();
      }, 1800);
    });
  })();
</script>
