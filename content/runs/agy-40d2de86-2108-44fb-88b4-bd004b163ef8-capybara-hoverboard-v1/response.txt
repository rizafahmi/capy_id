    reflGrad.addColorStop(0, state.boost ? 'rgba(255, 46, 147, 0.65)' : 'rgba(0, 246, 255, 0.55)');
    reflGrad.addColorStop(0.55, 'rgba(255, 62, 157, 0.25)');
    reflGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
    wCtx.fillStyle = reflGrad;
    wCtx.beginPath();
    wCtx.ellipse(rigX, groundY, 230 * shadowScale, 36 * shadowScale, 0, 0, Math.PI * 2);
    wCtx.fill();
    wCtx.restore();

    // 8. Collectible Floating Yuzus on Road
    updateAndDrawRoadSnacks(dt, rigX, groundY);
  }

  function drawMountains(horizonY, speedFactor, maxH, color, freq) {
    const scroll = state.time * 90 * speedFactor * state.speed;
    wCtx.fillStyle = color;
    wCtx.beginPath();
    wCtx.moveTo(0, horizonY);
    const step = 32;
    for (let x = 0; x <= width + step; x += step) {
      const wx = x + scroll;
      const h =
        Math.sin(wx * 0.0035 * freq) * (maxH * 0.55) +
        Math.cos(wx * 0.008 * freq) * (maxH * 0.35) +
        maxH * 0.6;
      wCtx.lineTo(x, horizonY - Math.max(8, h));
    }
    wCtx.lineTo(width, horizonY);
    wCtx.closePath();
    wCtx.fill();
  }

  function drawSkyline(horizonY, theme) {
    const scroll = (state.time * 48 * state.speed) % 2000;
    wCtx.save();
    wCtx.fillStyle = theme.mountainNear;
    for (const b of skyline) {
      let bx = (b.x - scroll) % 2000;
      if (bx < -120) bx += 2000;
      if (bx > width + 100) continue;
      wCtx.fillRect(bx, horizonY - b.h, b.w, b.h);

      // Glowing cyber windows
      wCtx.fillStyle = 'rgba(255, 214, 112, 0.28)';
      for (let wy = horizonY - b.h + 10; wy < horizonY - 12; wy += 14) {
        wCtx.fillRect(bx + 7, wy, b.w - 14, 4);
      }
      wCtx.fillStyle = theme.mountainNear;
    }
    wCtx.restore();
  }

  function updateAndDrawRoadSnacks(dt, capyX, capyGroundY) {
    for (let i = roadSnacks.length - 1; i >= 0; i--) {
      const s = roadSnacks[i];
      s.x -= dt * 390 * state.speed;
      s.bobPhase += dt * 5;

      const sy = capyGroundY - 52 + s.laneOffset * 0.45 + Math.sin(s.bobPhase) * 9;

      // Check pickup distance with Capybara
      const dx = s.x - (capyX + 95);
      const dy = sy - (capyGroundY - 55);
      if (!s.collected && Math.hypot(dx, dy) < 95) {
        s.collected = true;
        state.yuzuCount += 1;
        state.chillScore += 10;
        yuzuReadout.textContent = `🍊 ${state.yuzuCount}`;
        chillReadout.textContent = `${state.chillScore}%`;
        state.chewPhase = 1.2; // Happy munching animation!

        floatingTexts.push({
          text: '+1 YUZU! MUNCH~',
          x: s.x,
          y: sy - 25,
          vy: -1.8,
          life: 1
        });

        for (let k = 0; k < 14; k++) {
          const ang = (Math.PI * 2 * k) / 14;
          sparkParticles.push({
            x: s.x,
            y: sy,
            vx: Math.cos(ang) * 4.5,
            vy: Math.sin(ang) * 4.5,
            size: 4.5,
            color: '#ffd147',
            life: 1,
            decay: 0.035,
            star: true
          });
        }
      }

      if (s.collected) {
        s.scale -= dt * 6;
      }

      if (s.x < -80 || s.scale <= 0) {
        roadSnacks.splice(i, 1);
        if (roadSnacks.length < 2) {
          spawnRoadSnack(width + 250 + Math.random() * 450);
        }
        continue;
      }

      // Draw glowing collectible Yuzu on road
      wCtx.save();
      wCtx.translate(s.x, sy);
      wCtx.scale(s.scale, s.scale);

      // Aura
      wCtx.fillStyle = 'rgba(255, 212, 71, 0.28)';
      wCtx.beginPath();
      wCtx.arc(0, 0, 26, 0, Math.PI * 2);
      wCtx.fill();

      // Yuzu fruit
      wCtx.fillStyle = '#ffbe2e';
      wCtx.strokeStyle = '#6e3800';
      wCtx.lineWidth = 2.2;
      wCtx.beginPath();
      wCtx.ellipse(0, 0, 16, 13, 0, 0, Math.PI * 2);
      wCtx.fill();
      wCtx.stroke();

      // Green leaf
      wCtx.fillStyle = '#38d962';
      wCtx.beginPath();
      wCtx.ellipse(5, -13, 7, 4, -0.4, 0, Math.PI * 2);
      wCtx.fill();
      wCtx.restore();
    }
  }

  // Update Capybara & Hoverboard Articulated SVG Rig
  function updateCapybaraRig(dt) {
    // Autonomous gentle weaving if user hasn't moved mouse recently
    if (performance.now() - state.lastPointerMove > 2600) {
      state.pointerX = Math.sin(state.time * 0.75) * 0.42;
      state.pointerY = Math.cos(state.time * 1.1) * 0.25;
    }

    state.smoothX += (state.pointerX - state.smoothX) * Math.min(1, dt * 6);
    state.smoothY += (state.pointerY - state.smoothY) * Math.min(1, dt * 6);
    state.speed += (state.targetSpeed - state.speed) * Math.min(1, dt * 4);

    // Update Speedometer Readout
    const mph = Math.round(88 * state.speed + Math.sin(state.time * 7) * 2);
    speedReadout.textContent = `${mph} MPH`;

    // Harmonic Hover Oscillation
    const hoverBob = Math.sin(state.time * 4.4) * 11 + Math.cos(state.time * 8.8) * 2.5;
    const carveTilt = state.smoothX * 9 + Math.cos(state.time * 4.4) * 2.4 - (state.boost ? 4.5 : 0);

    // Trick / Jump Calculation
    let jumpY = 0;
    let trickRotation = 0;
    let yuzuPopY = 0;
    let yuzuRot = 0;
    let birdPopY = 0;

    if (state.trickActive) {
      state.trickProgress += dt * 1.55;
      if (state.trickProgress >= 1) {
        state.trickActive = false;
        state.trickProgress = 0;
      } else {
        const p = state.trickProgress;
        jumpY = -Math.sin(p * Math.PI) * 135;
        // Smooth ease-in-out 360 spin
        const ease = p < 0.5 ? 2 * p * p : 1 - Math.pow(-2 * p + 2, 2) / 2;
        trickRotation = ease * 360 * (state.trickType === 0 ? -1 : 1);
        // Yuzu and bird float slightly higher during peak of flip!
        yuzuPopY = -Math.sin(p * Math.PI) * 36;
        yuzuRot = ease * 360;
        birdPopY = -Math.sin(p * Math.PI) * 28;
      }
    }

    // Position the entire wrapper slightly with pointer
    const wrapper = document.getElementById('capy-rig-wrapper');
    const offsetX = state.smoothX * 95;
    const offsetY = state.smoothY * 32;
    wrapper.style.transform = `translate3d(${offsetX.toFixed(1)}px, ${offsetY.toFixed(1)}px, 0)`;

    // Apply Hover & Trick Transform to SVG Root
    const totalY = 300 + hoverBob + jumpY;
    const totalRot = carveTilt + trickRotation;
    rigRoot.setAttribute('transform', `translate(350, ${totalY.toFixed(2)}) rotate(${totalRot.toFixed(2)})`);

    // Subtle suspension compression between Capybara and Hoverboard
    const kneeFlex = Math.sin(state.time * 4.4 - 0.5) * 3.5;
    capybaraGroup.setAttribute('transform', `translate(-10, ${kneeFlex.toFixed(2)})`);

    // Head bob, chew animation & serene nod
    let headPitch = Math.sin(state.time * 4.4 + 0.6) * 2.2 + (state.boost ? 3.5 : 0);
    if (state.chewPhase > 0) {
      state.chewPhase = Math.max(0, state.chewPhase - dt);
      headPitch += Math.sin(state.time * 28) * 3.2;
      capyBlush.setAttribute('opacity', '0.68');
    } else {
      capyBlush.setAttribute('opacity', '0.32');
    }
    capyHead.setAttribute('transform', `translate(86, ${(-138 + kneeFlex * 0.4).toFixed(2)}) rotate(${headPitch.toFixed(2)})`);

    // Yuzu Hat balance physics
    const yuzuWobble = Math.sin(state.time * 5.5) * 4 + yuzuRot;
    const yuzuBob = Math.abs(Math.cos(state.time * 4.4)) * -3 + yuzuPopY;
    yuzuHat.setAttribute('transform', `translate(32, ${(-74 + yuzuBob).toFixed(2)}) rotate(${yuzuWobble.toFixed(2)})`);

    // Bird Buddy physics on rump
    const birdBob = Math.sin(state.time * 6.2 + 1) * 3 + birdPopY;
    const birdTilt = Math.cos(state.time * 5) * 5;
    birdBuddy.setAttribute('transform', `translate(-112, ${(-148 + birdBob).toFixed(2)}) rotate(${birdTilt.toFixed(2)})`);

    // Procedural Wind-Whipped Scarf Path Morphing
    const windFreq = state.time * (state.boost ? 18 : 11);
    const w1 = Math.sin(windFreq) * (state.boost ? 22 : 13);
    const w2 = Math.cos(windFreq - 1.3) * (state.boost ? 32 : 18);
    const w3 = Math.sin(windFreq - 2.4) * (state.boost ? 26 : 15);

    scarfPrimary.setAttribute(
      'd',
      `M 48,-52 C -20,${(-58 + w1).toFixed(1)} -95,${(-30 + w2).toFixed(1)} -168,${(-44 - w1).toFixed(1)} ` +
      `C -215,${(-54 + w3).toFixed(1)} -255,${(-34 + w2).toFixed(1)} -298,${(-46 + w3).toFixed(1)} ` +
      `L -282,${(-14 + w3).toFixed(1)} ` +
      `C -235,${(-6 + w2).toFixed(1)} -195,${(-26 - w1).toFixed(1)} -148,${(-16 + w2).toFixed(1)} ` +
      `C -80,${(-2 + w1).toFixed(1)} -18,-26 48,-24 Z`
    );

    scarfSecondary.setAttribute(
      'd',
      `M 42,-58 C -25,${(-68 - w1).toFixed(1)} -95,${(-45 + w3).toFixed(1)} -165,${(-62 + w2).toFixed(1)} ` +
      `C -205,${(-72 - w2).toFixed(1)} -245,${(-55 + w1).toFixed(1)} -278,${(-68 - w3).toFixed(1)} ` +
      `L -265,${(-40 - w3).toFixed(1)} ` +
      `C -225,${(-32 + w1).toFixed(1)} -185,${(-48 - w2).toFixed(1)} -145,${(-38 + w3).toFixed(1)} ` +
      `C -85,-22 -20,-42 42,-36 Z`
    );

    // Periodic Chill Blinking
    state.blinkTimer -= dt;
    if (state.blinkTimer <= 0) {
      if (!state.isBlinking) {
        state.isBlinking = true;
        state.blinkTimer = 0.16; // Duration of closed eye
        upperEyelid.setAttribute('d', 'M -14,7 Q 0,10 14,7 L 14,-11 L -14,-11 Z');
        eyelidLine.setAttribute('d', 'M -14,5 Q 1,9 15,5');
      } else {
        state.isBlinking = false;
        state.blinkTimer = 2.6 + Math.random() * 3.2;
        upperEyelid.setAttribute('d', 'M -14,-3 Q 0,-9 14,-2 L 14,-11 L -14,-11 Z');
        eyelidLine.setAttribute('d', 'M -14,-2 Q 1,-7 15,-1');
      }
    }

    // Periodic Cute Ear Flick
    state.earTwitchTimer -= dt;
    if (state.earTwitchTimer <= 0) {
      state.earTwitchAmount = 1;
      state.earTwitchTimer = 1.8 + Math.random() * 2.8;
    }
    if (state.earTwitchAmount > 0) {
      state.earTwitchAmount = Math.max(0, state.earTwitchAmount - dt * 6);
    }
    const earAngle = -Math.sin(state.earTwitchAmount * Math.PI * 2) * 18 - (state.boost ? 14 : 0);
    earNear.setAttribute('transform', `translate(-14, -52) rotate(${earAngle.toFixed(1)})`);
    earFar.setAttribute('transform', `translate(18, -56) rotate(${(earAngle * 0.7).toFixed(1)})`);

    // Hoverboard Anti-Gravity Repulsor Rings Pulse
    const pulse1 = (state.time * 2.2) % 1;
    const pulse2 = (state.time * 2.2 + 0.5) % 1;

    waveRear.setAttribute('rx', (32 + pulse1 * 42).toFixed(1));
    waveRear.setAttribute('ry', (8 + pulse1 * 11).toFixed(1));
    waveRear.setAttribute('cy', (38 + pulse1 * 28).toFixed(1));
    waveRear.setAttribute('opacity', ((1 - pulse1) * 0.75).toFixed(2));

    waveFront.setAttribute('rx', (32 + pulse1 * 42).toFixed(1));
    waveFront.setAttribute('ry', (8 + pulse1 * 11).toFixed(1));
    waveFront.setAttribute('cy', (38 + pulse1 * 28).toFixed(1));
    waveFront.setAttribute('opacity', ((1 - pulse1) * 0.75).toFixed(2));

    waveRear2.setAttribute('rx', (32 + pulse2 * 42).toFixed(1));
    waveRear2.setAttribute('ry', (8 + pulse2 * 11).toFixed(1));
    waveRear2.setAttribute('cy', (38 + pulse2 * 28).toFixed(1));
    waveRear2.setAttribute('opacity', ((1 - pulse2) * 0.65).toFixed(2));

    waveFront2.setAttribute('rx', (32 + pulse2 * 42).toFixed(1));
    waveFront2.setAttribute('ry', (8 + pulse2 * 11).toFixed(1));
    waveFront2.setAttribute('cy', (38 + pulse2 * 28).toFixed(1));
    waveFront2.setAttribute('opacity', ((1 - pulse2) * 0.65).toFixed(2));

    // Animated Sequential Side Chevrons
    const activeChev = Math.floor(state.time * 10 * state.speed) % chevrons.length;
    chevrons.forEach((el, idx) => {
      const dist = (idx - activeChev + chevrons.length) % chevrons.length;
      el.setAttribute('fill', dist === 0 ? '#00f6ff' : dist === 1 ? '#ff3e9d' : '#fff3a1');
      el.setAttribute('opacity', dist <= 1 ? '1' : '0.45');
    });
  }

  // Foreground FX Canvas: Thruster Plasma Jet, Sparks & Speed Lines
  function updateAndDrawFX(dt) {
    fCtx.clearRect(0, 0, width, height);
    const theme = THEMES[state.themeIndex];

    // Get exact screen coordinates of Hoverboard thrusters for pixel-locked exhaust
    const rearRect = thrusterRearCore.getBoundingClientRect();
    const frontRect = thrusterFrontCore.getBoundingClientRect();

    const emitPoints = [
      { x: rearRect.left + rearRect.width * 0.5, y: rearRect.top + rearRect.height * 0.5, main: true },
      { x: frontRect.left + frontRect.width * 0.5, y: frontRect.top + frontRect.height * 0.5, main: false }
    ];

    // Spawn new plasma exhaust particles
    const countPerFrame = state.boost ? 5 : 3;
    for (const pt of emitPoints) {
      for (let i = 0; i < countPerFrame; i++) {
        thrusterParticles.push({
          x: pt.x + (Math.random() - 0.5) * 26,
          y: pt.y + (Math.random() - 0.5) * 8,
          vx: -(6 + Math.random() * 7) * state.speed,
          vy: (Math.random() - 0.35) * 2.2,
          r: (pt.main ? 10 : 7) + Math.random() * 8,
          color: Math.random() > 0.38 ? theme.thrusterA : theme.thrusterB,
          life: 1,
          decay: 0.032 + Math.random() * 0.025
        });
      }
    }

    // Render plasma trail with additive glow
    fCtx.save();
    fCtx.globalCompositeOperation = 'lighter';
    for (let i = thrusterParticles.length - 1; i >= 0; i--) {
      const p = thrusterParticles[i];
      p.x += p.vx * dt * 60;
      p.y += p.vy * dt * 60;
      p.life -= p.decay * dt * 60;

      if (p.life <= 0) {
        thrusterParticles.splice(i, 1);
        continue;
      }

      const radius = Math.max(1, p.r * p.life);
      const grad = fCtx.createRadialGradient(p.x, p.y, 0, p.x, p.y, radius * 1.6);
      grad.addColorStop(0, '#ffffff');
      grad.addColorStop(0.35, p.color);
      grad.addColorStop(1, 'rgba(0, 0, 0, 0)');

      fCtx.fillStyle = grad;
      fCtx.globalAlpha = p.life * 0.85;
      fCtx.beginPath();
      fCtx.arc(p.x, p.y, radius * 1.6, 0, Math.PI * 2);
      fCtx.fill();
    }
    fCtx.restore();

    // Render Trick & Snack Star Particles
    for (let i = sparkParticles.length - 1; i >= 0; i--) {
      const s = sparkParticles[i];
      s.x += s.vx * dt * 60;
      s.y += s.vy * dt * 60;
      s.vy += 0.12 * dt * 60;
      s.life -= s.decay * dt * 60;

      if (s.life <= 0) {
        sparkParticles.splice(i, 1);
        continue;
      }

      fCtx.save();
      fCtx.translate(s.x, s.y);
      fCtx.rotate(state.time * 6 + i);
      fCtx.globalAlpha = Math.max(0, s.life);
      fCtx.fillStyle = s.color;
      drawStar(fCtx, 0, 0, 4, s.size, s.size * 0.42);
      fCtx.restore();
    }

    // Render Hyper-Boost Anime Speed Lines
    if (state.speed > 1.25) {
      const boostAlpha = Math.min(1, (state.speed - 1.2) * 0.9);
      fCtx.save();
      fCtx.strokeStyle = '#ffffff';
      fCtx.lineWidth = 2;
      for (const sl of speedLines) {
        sl.x -= sl.speed * state.speed * dt * 60;
        if (sl.x + sl.len < 0) {
          sl.x = width + Math.random() * 200;
          sl.y = Math.random() * height;
        }
        fCtx.globalAlpha = sl.alpha * boostAlpha;
        fCtx.beginPath();
        fCtx.moveTo(sl.x, sl.y);
        fCtx.lineTo(sl.x + sl.len, sl.y);
        fCtx.stroke();
      }
      fCtx.restore();
    }

    // Render Floating Score Popups
    for (let i = floatingTexts.length - 1; i >= 0; i--) {
      const ft = floatingTexts[i];
      ft.y += ft.vy * dt * 60;
      ft.life -= 0.018 * dt * 60;
      if (ft.life <= 0) {
        floatingTexts.splice(i, 1);
        continue;
      }
      fCtx.save();
      fCtx.font = '800 18px -apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif';
      fCtx.textAlign = 'center';
      fCtx.globalAlpha = Math.min(1, ft.life * 1.4);
      fCtx.shadowColor = 'rgba(0, 0, 0, 0.8)';
      fCtx.shadowBlur = 8;
      fCtx.fillStyle = '#ffe373';
      fCtx.fillText(ft.text, ft.x, ft.y);
      fCtx.restore();
    }
  }

  function drawStar(ctx, cx, cy, spikes, outerR, innerR) {
    let rot = (Math.PI / 2) * 3;
    let x = cx;
    let y = cy;
    const step = Math.PI / spikes;
    ctx.beginPath();
    ctx.moveTo(cx, cy - outerR);
    for (let i = 0; i < spikes; i++) {
      x = cx + Math.cos(rot) * outerR;
      y = cy + Math.sin(rot) * outerR;
      ctx.lineTo(x, y);
      rot += step;

      x = cx + Math.cos(rot) * innerR;
      y = cy + Math.sin(rot) * innerR;
      ctx.lineTo(x, y);
      rot += step;
    }
    ctx.lineTo(cx, cy - outerR);
    ctx.closePath();
    ctx.fill();
  }

  // Main 60fps Animation Loop
  let lastTime = performance.now();
  function animate(now) {
    const dt = Math.min(0.05, (now - lastTime) / 1000);
    lastTime = now;
    state.time += dt;

    updateCapybaraRig(dt);
    drawWorld(dt);
    updateAndDrawFX(dt);

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
})();
