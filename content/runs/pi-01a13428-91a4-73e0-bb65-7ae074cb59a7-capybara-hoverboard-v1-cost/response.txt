<script>
(function () {
  "use strict";

  var scene  = document.getElementById("scene");
  var canvas = document.getElementById("fx");
  var ctx    = canvas.getContext("2d");
  var craft  = document.querySelector(".craft");

  var W = 0, H = 0, scale = 1, ox = 0, oy = 0, dpr = 1;
  var craftOffset = 0;

  function resize() {
    var rect = scene.getBoundingClientRect();
    dpr = Math.min(window.devicePixelRatio || 1, 2);
    W = rect.width;
    H = rect.height;
    canvas.width  = Math.max(1, Math.round(W * dpr));
    canvas.height = Math.max(1, Math.round(H * dpr));
    canvas.style.width  = W + "px";
    canvas.style.height = H + "px";
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    scale = Math.min(W / 800, H / 500);
    ox = (W - 800 * scale) / 2;
    oy = (H - 500 * scale) / 2;
  }

  function toScreen(x, y) {
    return [ox + x * scale, oy + y * scale];
  }

  var COLORS = ["#5ff2ff", "#9bffff", "#ff6ad5", "#ffffff", "#7aa8ff"];
  var parts = [];

  function spawnSpark() {
    var p = toScreen(295 + Math.random() * 210, 407 + craftOffset);
    parts.push({
      x: p[0],
      y: p[1],
      vx: (Math.random() - 0.5) * 55 * scale,
      vy: (35 + Math.random() * 80) * scale,
      life: 0,
      max: 0.55 + Math.random() * 0.7,
      r: (1.1 + Math.random() * 2.3) * scale,
      c: COLORS[(Math.random() * COLORS.length) | 0],
      kind: "spark"
    });
  }

  function spawnDust() {
    var p = toScreen(Math.random() * 800, 440 + Math.random() * 60);
    parts.push({
      x: p[0],
      y: p[1],
      vx: (Math.random() - 0.5) * 12 * scale,
      vy: -(8 + Math.random() * 24) * scale,
      life: 0,
      max: 2 + Math.random() * 2.6,
      r: (0.6 + Math.random() * 1.3) * scale,
      c: "#7fdcff",
      kind: "dust"
    });
  }

  function readCraftOffset() {
    var t = getComputedStyle(craft).transform;
    if (!t || t === "none") { craftOffset = 0; return; }
    var m = t.match(/matrix\(([^)]+)\)/);
    if (m) {
      var v = m[1].split(",");
      var f = parseFloat(v[5]);
      craftOffset = isNaN(f) ? 0 : f;
    }
  }

  var last = performance.now();
  var sparkAcc = 0;
  var dustAcc = 0;

  function frame(now) {
    var dt = Math.min(0.05, (now - last) / 1000);
    last = now;

    readCraftOffset();

    ctx.clearRect(0, 0, W, H);
    ctx.globalCompositeOperation = "lighter";

    sparkAcc += dt;
    while (sparkAcc > 0.02) { sparkAcc -= 0.02; spawnSpark(); }

    dustAcc += dt;
    if (dustAcc > 0.14) { dustAcc = 0; spawnDust(); }

    for (var i = parts.length - 1; i >= 0; i--) {
      var p = parts[i];
      p.life += dt;
      if (p.life >= p.max) { parts.splice(i, 1); continue; }

      p.x += p.vx * dt;
      p.y += p.vy * dt;

      if (p.kind === "spark") {
        p.vy += 70 * scale * dt;
        p.vx *= (1 - 1.7 * dt);
      } else {
        p.vx += Math.sin(p.y * 0.06 + p.life * 3) * 7 * scale * dt;
      }

      var t = p.life / p.max;
      var alpha = (1 - t) * (p.kind === "spark" ? 1 : 0.45);
      var radius = p.r * (p.kind === "spark" ? 4 : 3.4);

      var g = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, radius);
      g.addColorStop(0, p.c);
      g.addColorStop(1, "rgba(0,0,0,0)");

      ctx.globalAlpha = alpha;
      ctx.fillStyle = g;
