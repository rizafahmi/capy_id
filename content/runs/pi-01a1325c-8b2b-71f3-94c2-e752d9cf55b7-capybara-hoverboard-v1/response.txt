<script>
(function(){
  var stage = document.getElementById('stage');
  var starsBox = document.getElementById('stars');
  var city = document.getElementById('city');
  var canvas = document.getElementById('fx');
  var ctx = canvas.getContext('2d');
  var eye = document.getElementById('eyeGroup');
  var floatGroup = document.getElementById('floatGroup');
  var heroWrap = document.getElementById('heroWrap');
  var meterFill = document.getElementById('meterFill');
  var statusEl = document.getElementById('status');

  // stars
  var starCount = 110;
  for(var i=0;i<starCount;i++){
    var s=document.createElement('div');
    s.className='star';
    var sz = Math.random()*2.4+0.8;
    s.style.width=sz+'px';s.style.height=sz+'px';
    s.style.left=(Math.random()*100)+'%';
    s.style.top=(Math.random()*100)+'%';
    s.style.setProperty('--tw',(Math.random()*3+1.5)+'s');
    s.style.animationDelay=(Math.random()*3)+'s';
    if(Math.random()<.18){s.style.background='#7df9ff';s.style.boxShadow='0 0 8px #7df9ff';}
    if(Math.random()<.12){s.style.background='#ffb14e';s.style.boxShadow='0 0 8px #ffb14e';}
    starsBox.appendChild(s);
  }
  // city buildings
  var widths=[46,62,38,74,52,88,44,66,58,40,70,48,60,36,80,54,62,44];
  widths.forEach(function(w,i){
    var b=document.createElement('div');
    b.className='bldg'+(i%3===0?' blink':'');
    b.style.width=w+'px';
    b.style.height=(40+Math.abs(Math.sin(i*1.7))*95+Math.random()*30)+'px';
    city.appendChild(b);
  });

  // canvas sizing
  var W=0,H=0,DPR=1;
  function resize(){
    DPR=Math.min(window.devicePixelRatio||1,2);
    W=stage.clientWidth;H=stage.clientHeight;
    canvas.width=W*DPR;canvas.height=H*DPR;
    canvas.style.width=W+'px';canvas.style.height=H+'px';
    ctx.setTransform(DPR,0,0,DPR,0,0);
  }
  window.addEventListener('resize',resize);resize();

  var mouseX=0.5, mouseY=0.5, targetX=0.5, targetY=0.5;
  var boost=0, boostTarget=0;
  window.addEventListener('pointermove',function(e){
    targetX=e.clientX/window.innerWidth;
    targetY=e.clientY/window.innerHeight;
  });
  function startBoost(){boostTarget=1;document.body.classList.add('boosting');statusEl.textContent='BOOST!!';}
  function endBoost(){boostTarget=0;document.body.classList.remove('boosting');statusEl.textContent='CRUISING';}
  window.addEventListener('pointerdown',startBoost);
  window.addEventListener('pointerup',endBoost);
  window.addEventListener('keydown',function(e){if(e.code==='Space'){e.preventDefault();startBoost();}});
  window.addEventListener('keyup',function(e){if(e.code==='Space')endBoost();});

  // blink loop
  (function blinkLoop(){
    setTimeout(function(){
      eye.classList.add('blink');
      setTimeout(function(){eye.classList.remove('blink');},130);
      // double blink sometimes
      if(Math.random()<.18){
        setTimeout(function(){
          eye.classList.add('blink');
          setTimeout(function(){eye.classList.remove('blink');},120);
        },220);
      }
      setTimeout(blinkLoop, 2200+Math.random()*2600);
    }, 1800+Math.random()*2000);
  })();

  // particles
  var sparks=[], dust=[], lines=[];
  for(var d=0;d<45;d++){
    dust.push({x:Math.random(),y:Math.random(),r:Math.random()*2+0.6,s:Math.random()*0.0006+0.0002,o:Math.random()*.6+.2,ph:Math.random()*6.28});
  }
  function heroAnchor(){
    var r=heroWrap.getBoundingClientRect();
    var sr=stage.getBoundingClientRect();
    return {
      x1:r.left-sr.left+r.width*0.335,
      x2:r.left-sr.left+r.width*0.665,
      y:r.top-sr.top+r.height*0.74
    };
  }
  function spawnSpark(a, big){
    var fromLeft = Math.random()<.5;
    var bx = fromLeft? a.x1 : a.x2;
    var hue = fromLeft? 185+Math.random()*15 : 315+Math.random()*20;
    sparks.push({
      x:bx+(Math.random()-.5)*18,
      y:a.y+(Math.random()-.5)*8,
      vx:(Math.random()-.5)*1.2 -1.2 -boost*4,
      vy:Math.random()*2.2+0.8+boost*1.5,
      life:1,
      decay:0.015+Math.random()*0.02,
      size:(big?3:1.5)+Math.random()*2.5+boost*1.5,
      hue:hue,
      wob:Math.random()*6.28
    });
  }

  var t=0;
  function frame(){
    t+=0.016;
    mouseX += (targetX-mouseX)*0.06;
    mouseY += (targetY-mouseY)*0.06;
    boost += (boostTarget-boost)*0.08;

    // parallax + steer
    var px=(mouseX-0.5), py=(mouseY-0.5);
    heroWrap.style.transform='translate(calc(-50% + '+(px*36)+'px), calc(-50% + '+(py*22 - boost*26)+'px)) rotate('+(px*4)+'deg)';
    starsBox.style.transform='translate('+(px*-14)+'px,'+(py*-8)+'px)';
    city.style.transform='translateX('+(px*-22 - boost*10)+'px)';
    document.querySelector('.moon').style.transform='translate('+(px*-20)+'px,'+(py*-12)+'px)';

    // eye follow
    if(eye) eye.style.translate=((px*6)+'px '+ (py*4)+'px)';

    meterFill.style.width=(18+boost*82)+'%';
    statusEl.textContent = boost>0.5?'BOOST!!':(boost>0.15?'CHARGING':'CRUISING');

    var a=heroAnchor();
    var rate = 2+Math.floor(boost*10);
    for(var k=0;k<rate;k++) spawnSpark(a, boost>0.4);
    if(sparks.length>420) sparks.splice(0,sparks.length-420);

    // speedlines
    if((boost>0.3||Math.abs(px)>0.28) && Math.random()<.5){
      lines.push({x:W*0.9+Math.random()*80,y:Math.random()*H*0.8+H*0.05,len:80+Math.random()*160+boost*200,v:12+Math.random()*10+boost*22,o:.5});
    }

    ctx.clearRect(0,0,W,H);

    // dust / fireflies
    ctx.save();
    for(var i=0;i<dust.length;i++){
      var p=dust[i];
      p.y-=p.s*(1+boost*4);
      p.x-=p.s*0.6*(1+boost*6)+px*0.0008;
      if(p.y<-0.05){p.y=1.05;p.x=Math.random();}
      if(p.x<-0.05)p.x=1.05;
      var dx=p.x*W, dy=p.y*H;
      var tw=.4+.6*Math.abs(Math.sin(t*2+p.ph));
      ctx.globalAlpha=p.o*tw;
      ctx.fillStyle= i%4===0 ? '#7df9ff' : (i%5===0?'#ffb14e':'#ffffff');
      ctx.beginPath();ctx.arc(dx,dy,p.r,0,6.283);ctx.fill();
    }
    ctx.restore();

    // thruster sparks
    for(var j=sparks.length-1;j>=0;j--){
      var sp=sparks[j];
      sp.x+=sp.vx+Math.sin(t*10+sp.wob)*0.6;
      sp.y+=sp.vy;
      sp.vy*=0.985; sp.vx*=0.985;
      sp.life-=sp.decay;
      if(sp.life<=0){sparks.splice(j,1);continue;}
      var alpha=Math.max(0,sp.life);
      ctx.globalAlpha=alpha;
      // glow
      ctx.fillStyle='hsla('+sp.hue+',100%,65%,'+(alpha*0.35)+')';
      ctx.beginPath();ctx.arc(sp.x,sp.y,sp.size*2.4,0,6.283);ctx.fill();
      ctx.fillStyle='hsl('+sp.hue+',100%,'+(70+alpha*20)+'%)';
      ctx.beginPath();ctx.arc(sp.x,sp.y,sp.size*alpha+0.4,0,6.283);ctx.fill();
      ctx.fillStyle='rgba(255,255,255,'+(alpha*0.9)+')';
      ctx.beginPath();ctx.arc(sp.x,sp.y,Math.max(0.4,sp.size*alpha*0.35),0,6.283);ctx.fill();
    }
    ctx.globalAlpha=1;

    // speed lines
    ctx.save();
    ctx.lineCap='round';
    for(var l=lines.length-1;l>=0;l--){
      var L=lines[l];
      L.x-=L.v*(1+boost*1.2);
      L.o-=0.02;
      if(L.o<=0||L.x+L.len<0){lines.splice(l,1);continue;}
      var grad=ctx.createLinearGradient(L.x,0,L.x+L.len,0);
      grad.addColorStop(0,'rgba(255,255,255,0)');
      grad.addColorStop(1,'rgba(255,255,255,'+L.o+')');
      ctx.strokeStyle=grad;
      ctx.lineWidth=2+(boost*1.5);
      ctx.beginPath();ctx.moveTo(L.x,L.y);ctx.lineTo(L.x+L.len,L.y);ctx.stroke();
    }
    ctx.restore();

    // hover glow flicker under board drawn on canvas for extra life
    var gx=(a.x1+a.x2)/2, gy=a.y+52;
    var gr=90+Math.sin(t*7)*8+boost*30;
    var rg=ctx.createRadialGradient(gx,gy,0,gx,gy,gr*1.6);
    rg.addColorStop(0,'rgba(0,255,255,'+(0.22+boost*0.25)+')');
    rg.addColorStop(0.5,'rgba(255,45,150,'+(0.12+boost*0.15)+')');
    rg.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=rg;
    ctx.fillRect(gx-gr*1.6,gy-20,gr*3.2,70);

    requestAnimationFrame(frame);
  }
  frame();
})();
</script>
</body>
