      function animate(now) {

        state.time += dt;

        if (keys['KeyW'] || keys['ArrowUp'])    { state.targetY -= keySpeed; }
        if (keys['KeyS'] || keys['ArrowDown'])  { state.targetY += keySpeed; }

        state.targetY = Math.max(height * 0.25, Math.min(height * 0.88, state.targetY));

        // 3. SMOOTH CHARACTER MOVEMENT
        const followSpeed = 6.0;
        state.vx = (state.targetX - state.x) * followSpeed;
        state.vy = (state.targetY - state.y) * followSpeed;
        state.x += state.vx * dt;
        state.y += state.vy * dt;

        // Hover float harmonic oscillation
        const hoverFloat = Math.sin(state.time * 3.2) * 10 + Math.sin(state.time * 6.5) * 3;
        const currentY = state.y + hoverFloat;

        // Banking tilt
        const targetRoll = (state.vx / 350) * 14 + Math.sin(state.time * 2.3) * 2;
        const targetPitch = (state.vy / 300) * 8 - (state.isTurbo ? 10 : 0) + Math.cos(state.time * 2.7) * 2;
        state.roll += (targetRoll - state.roll) * 8 * dt;
        state.pitch += (targetPitch - state.pitch) * 8 * dt;

        // Speed interpolation
        state.currentSpeed += (state.targetSpeed - state.currentSpeed) * 4 * dt;
        speedInd.textContent = Math.round(state.currentSpeed);

        const speedRatio = state.currentSpeed / state.baseSpeed;
        state.distance += state.currentSpeed * 2.5 * dt;

        // 4. TRICK JUMP & ROTATION CALCULATION
        let trickTransform = '';
        if (state.isTricking) {
          state.trickProgress += dt / state.trickDuration;
          if (state.trickProgress >= 1.0) {
            state.isTricking = false;
            state.trickProgress = 0;
            state.trickJumpH = 0;
          } else {
            // Jump parabolic arc
            state.trickJumpH = Math.sin(state.trickProgress * Math.PI) * 110;
            const p = state.trickProgress;
            if (state.trickStyle === 0) {
              // 360 Kickflip
              trickTransform = ` rotateX(${p * 360}deg)`;
            } else if (state.trickStyle === 1) {
              // Barrel roll
              trickTransform = ` rotateZ(${-p * 360}deg)`;
            } else {
              // Pop spin
              trickTransform = ` rotateY(${p * 360}deg)`;
            }
          }
        }

        // Responsive character scaling for phones/tablets
        const charScale = Math.min(1.0, Math.max(0.68, width / 750));

        // Apply 3D CSS Transform to character container
        const totalY = currentY - state.trickJumpH;
        charRoot.style.transform = `translate3d(${state.x}px, ${totalY}px, 0px) scale(${charScale}) rotateZ(${state.roll}deg) rotateX(${state.pitch}deg)${trickTransform}`;

        // 5. YUZU FRUIT SPRING DYNAMICS

        state.yuzuRotV += (targetYuzuAngle - state.yuzuRot) * 22 * dt;
        state.yuzuRotV *= 0.82;
        state.yuzuRot += state.yuzuRotV * dt * 25;

          state.earTwitchTimer += dt * 8;
          state.earAngle = Math.sin(state.earTwitchTimer * Math.PI * 4) * 14 * Math.max(0, 1 - state.earTwitchTimer);
          if (state.earTwitchTimer >= 1.0) {
            state.earTwitchTimer = 0;
            state.earAngle = 0;
            state.nextEarTwitch = state.time + 3.0 + Math.random() * 4.0;

          capyEar.setAttribute('transform', `rotate(${state.earAngle.toFixed(1)} 300 105)`);

        const nozzleOffsetX = -190;
        const nozzleOffsetY = 40;
        const flameWorldX = state.x + Math.cos(rad) * nozzleOffsetX - Math.sin(rad) * nozzleOffsetY;
        const flameWorldY = totalY + Math.sin(rad) * nozzleOffsetX + Math.cos(rad) * nozzleOffsetY;

          createSpeedLine();
        }

        // RENDER BACKGROUND CANVAS
        // ==========================================

        const horizonY = height * 0.58;

        skyGrad.addColorStop(0.0, '#060312');
        skyGrad.addColorStop(0.35, '#1e0836');
        skyGrad.addColorStop(0.68, '#581159');
        skyGrad.addColorStop(0.88, '#e62f6b');
        skyGrad.addColorStop(1.0, '#ffa83b');
        bgCtx.fillStyle = skyGrad;
        bgCtx.fillRect(0, 0, width, horizonY);

        // B. Stars
        bgCtx.save();
        for (let i = 0; i < STARS.length; i++) {
          const s = STARS[i];
          const starX = (s.x * width);
          const starY = (s.y * horizonY);
          const flicker = Math.sin(state.time * s.speed + i) * 0.3 + 0.7;
          bgCtx.fillStyle = `rgba(255, 255, 255, ${s.alpha * flicker})`;
          bgCtx.beginPath();
          bgCtx.arc(starX, starY, s.r, 0, Math.PI * 2);
          bgCtx.fill();
        }
        bgCtx.restore();

        // C. Giant Synthwave Sun
        const sunRadius = Math.min(width, height) * 0.22;
        const sunX = width * 0.5;
        const sunY = horizonY - 15;

        bgCtx.save();
        bgCtx.beginPath();
        bgCtx.arc(sunX, sunY, sunRadius, 0, Math.PI * 2);
        bgCtx.clip();

        const sunGrad = bgCtx.createLinearGradient(0, sunY - sunRadius, 0, sunY + sunRadius);
        sunGrad.addColorStop(0.0, '#fff677');
        sunGrad.addColorStop(0.4, '#ff3b75');
        sunGrad.addColorStop(1.0, '#7928ca');
        bgCtx.fillStyle = sunGrad;
        bgCtx.fillRect(sunX - sunRadius, sunY - sunRadius, sunRadius * 2, sunRadius * 2);

        // Synthwave Horizontal Dark Slices
        const sliceCount = 14;
        for (let i = 0; i < sliceCount; i++) {
          const sliceNorm = i / sliceCount;
          const sliceY = (sunY - sunRadius * 0.1) + sliceNorm * (sunRadius * 1.1);
          const sliceH = Math.pow(sliceNorm, 1.8) * 9 + 2;
          bgCtx.fillStyle = '#0e041d';
          bgCtx.fillRect(sunX - sunRadius - 10, sliceY, sunRadius * 2 + 20, sliceH);
        }
        bgCtx.restore();

        // Sun Corona Glow
        const corona = bgCtx.createRadialGradient(sunX, sunY, sunRadius * 0.8, sunX, sunY, sunRadius * 1.4);
        corona.addColorStop(0.0, 'rgba(255, 59, 117, 0.35)');
        corona.addColorStop(1.0, 'rgba(255, 59, 117, 0.0)');
        bgCtx.fillStyle = corona;
        bgCtx.beginPath();
        bgCtx.arc(sunX, sunY, sunRadius * 1.4, 0, Math.PI * 2);
        bgCtx.fill();

        // D. Distant Mountain Silhouettes (Parallax)
        state.mountainOffset = (state.mountainOffset + state.currentSpeed * 0.05 * dt) % (width * 1.5);
        bgCtx.save();
        bgCtx.fillStyle = '#17092c';
        bgCtx.strokeStyle = '#ff00aa';
        bgCtx.lineWidth = 1.5;
        bgCtx.beginPath();
        bgCtx.moveTo(0, horizonY);
        const mSteps = 20;
        for (let i = 0; i <= mSteps; i++) {
          const mx = (i / mSteps) * width;
          const noise = Math.sin((mx + state.mountainOffset) * 0.005) * 45 + Math.cos((mx + state.mountainOffset) * 0.012) * 25;
          const my = horizonY - 40 - Math.abs(noise);
          bgCtx.lineTo(mx, my);
        }
        bgCtx.lineTo(width, horizonY);
        bgCtx.closePath();
        bgCtx.fill();
        bgCtx.stroke();
        bgCtx.restore();

        // E. Cyber Skyline with Glowing Windows & Beacons
        state.skylineOffset = (state.skylineOffset + state.currentSpeed * 0.18 * dt) % totalSkylineWidth;
        bgCtx.save();
        let curX = -state.skylineOffset;
        while (curX < width + 100) {
          for (let i = 0; i < BUILDINGS.length; i++) {
            const b = BUILDINGS[i];
            const bx = curX;
            const by = horizonY - b.h;

            if (bx + b.w > -50 && bx < width + 50) {
              // Building body
              bgCtx.fillStyle = b.color;
              bgCtx.fillRect(bx, by, b.w, b.h);

              // Building top rim glow
              bgCtx.fillStyle = '#00f5ff';
              bgCtx.fillRect(bx, by, b.w, 1.5);

              // Windows
              for (let w = 0; w < b.windows.length; w++) {
                const win = b.windows[w];
                bgCtx.fillStyle = win.color;
                bgCtx.globalAlpha = win.alpha;
                bgCtx.fillRect(bx + win.x, by + win.y, 5, 6);
                bgCtx.globalAlpha = 1.0;
              }

              // Antenna strobe
              if (b.hasAntenna) {
                const antX = bx + b.w * 0.5;
                bgCtx.strokeStyle = '#2b1b48';
                bgCtx.lineWidth = 2;
                bgCtx.beginPath();
                bgCtx.moveTo(antX, by);
                bgCtx.lineTo(antX, by - b.antennaH);
                bgCtx.stroke();

                const blink = Math.sin(state.time * 6 + i) > 0.2;
                bgCtx.fillStyle = blink ? '#ff0055' : '#ffffff';
                bgCtx.beginPath();
                bgCtx.arc(antX, by - b.antennaH, 2.5, 0, Math.PI * 2);
                bgCtx.fill();
              }
            }
            curX += b.w + 12;
          }
        }
        bgCtx.restore();

        // F. Perspective Cyber Highway (3D Road)
        const roadH = height - horizonY;
        const roadGrad = bgCtx.createLinearGradient(0, horizonY, 0, height);
        roadGrad.addColorStop(0.0, '#090514');
        roadGrad.addColorStop(0.4, '#150a24');
        roadGrad.addColorStop(1.0, '#0a0514');
        bgCtx.fillStyle = roadGrad;
        bgCtx.fillRect(0, horizonY, width, roadH);

        // Perspective Ground Grid & Highway Lines
        const vpX = width * 0.5;
        const cameraH = roadH;
        const numTransverse = 22;

        // 1. Transverse Cross Grid Lines (moving forward)
        bgCtx.save();
        for (let i = 0; i < numTransverse; i++) {
          const zProgress = ((state.distance * 0.0035 + i / numTransverse) % 1.0);
          const z = Math.pow(zProgress, 2.2); // Exponential depth
          const lineY = horizonY + z * roadH;

          if (lineY > horizonY + 2) {
            const alpha = Math.min(1.0, z * 1.5);
            bgCtx.strokeStyle = `rgba(0, 245, 255, ${alpha * 0.45})`;
            bgCtx.lineWidth = Math.max(1, z * 2.8);
            bgCtx.beginPath();
            bgCtx.moveTo(0, lineY);
            bgCtx.lineTo(width, lineY);
            bgCtx.stroke();
          }
        }

        // 2. Longitudinal Perspective Radiating Grid Lines
        const numLong = 18;
        for (let i = -numLong / 2; i <= numLong / 2; i++) {
          const bottomX = vpX + i * (width / (numLong * 0.65));
          bgCtx.strokeStyle = (i === 0) ? 'rgba(255, 0, 127, 0.7)' : 'rgba(0, 245, 255, 0.28)';
          bgCtx.lineWidth = (i === 0) ? 2.5 : 1.2;
          bgCtx.beginPath();
          bgCtx.moveTo(vpX, horizonY);
          bgCtx.lineTo(bottomX, height);
          bgCtx.stroke();
        }

        // 3. Center High-Speed Dashed Divider Strip
        const dashCount = 14;
        for (let i = 0; i < dashCount; i++) {
          const zProgress = ((state.distance * 0.006 + i / dashCount) % 1.0);
          const z = Math.pow(zProgress, 2.5);
          const dashY = horizonY + z * roadH;
          const dashLen = Math.max(4, z * 45);
          const dashW = Math.max(2, z * 7);

          if (dashY > horizonY + 4) {
            bgCtx.fillStyle = `rgba(255, 235, 80, ${z * 0.9})`;
            bgCtx.fillRect(vpX - dashW / 2, dashY, dashW, dashLen);
          }
        }

        // 4. Outer Highway Neon Rails
        const leftRailBottom = 0;
        const rightRailBottom = width;
        bgCtx.strokeStyle = '#ff00aa';
        bgCtx.lineWidth = 3;
        bgCtx.beginPath();
        bgCtx.moveTo(vpX, horizonY);
        bgCtx.lineTo(leftRailBottom, height);
        bgCtx.moveTo(vpX, horizonY);
        bgCtx.lineTo(rightRailBottom, height);
        bgCtx.stroke();
        bgCtx.restore();

        // G. Ground Projection & Reflection of Hoverboard
        // Position on asphalt directly below the hoverboard
        const groundY = Math.min(height - 20, Math.max(horizonY + 30, state.y + 45));
        const altitude = Math.max(10, groundY - state.y);
        const shadowScale = Math.max(0.5, 1.4 - (altitude / 250));
        const shadowAlpha = Math.max(0.2, 0.85 - (altitude / 260));

        bgCtx.save();
        bgCtx.translate(state.x, groundY);
        bgCtx.scale(shadowScale, shadowScale * 0.28);

        // Outer Neon Glow Pool
        const groundGrad = bgCtx.createRadialGradient(0, 0, 10, 0, 0, 160);
        groundGrad.addColorStop(0.0, `rgba(0, 245, 255, ${shadowAlpha * 0.9})`);
        groundGrad.addColorStop(0.35, `rgba(255, 0, 127, ${shadowAlpha * 0.5})`);
        groundGrad.addColorStop(1.0, 'rgba(0, 245, 255, 0.0)');
        bgCtx.fillStyle = groundGrad;
        bgCtx.beginPath();
        bgCtx.arc(0, 0, 160, 0, Math.PI * 2);
        bgCtx.fill();

        // Expanding repulsor shockwave ripples
        const rippleCount = 3;
        for (let r = 0; r < rippleCount; r++) {
          const ripProgress = (((state.time * 2.2 + r / rippleCount) % 1.0) + 1.0) % 1.0;
          const ripRadius = Math.max(0.1, ripProgress * 150);
          const ripAlpha = (1.0 - ripProgress) * shadowAlpha * 0.7;
          bgCtx.strokeStyle = `rgba(0, 245, 255, ${ripAlpha})`;
          bgCtx.lineWidth = 3;
          bgCtx.beginPath();
          bgCtx.arc(0, 0, ripRadius, 0, Math.PI * 2);
          bgCtx.stroke();
        }
        bgCtx.restore();

        // H. Render Thruster Plasma Particles
        bgCtx.save();
        bgCtx.globalCompositeOperation = 'lighter';
        for (let i = thrusterParticles.length - 1; i >= 0; i--) {
          const p = thrusterParticles[i];
          p.x += p.vx * dt;
          p.y += p.vy * dt;
          p.life -= p.decay * dt;

          if (p.life <= 0) {
            thrusterParticles.splice(i, 1);
            continue;
          }

          const currentR = Math.max(0.1, p.r * p.life);
          const alpha = p.life * 0.85;
          let col = `rgba(0, 245, 255, ${alpha})`;
          if (p.colorType > 0.6) col = `rgba(255, 0, 127, ${alpha})`;
          else if (p.colorType > 0.35) col = `rgba(255, 255, 255, ${alpha})`;

          bgCtx.fillStyle = col;
          bgCtx.beginPath();
          bgCtx.arc(p.x, p.y, currentR, 0, Math.PI * 2);
          bgCtx.fill();
        }
        bgCtx.restore();

        // ==========================================
        // RENDER FOREGROUND CANVAS (Speed Lines, Trick Sparks)
        // ==========================================
        fgCtx.clearRect(0, 0, width, height);

        // A. Speed Lines
        fgCtx.save();
        for (let i = speedLines.length - 1; i >= 0; i--) {
          const sl = speedLines[i];
          sl.x -= sl.speed * dt;
          if (sl.x + sl.len < 0) {
            speedLines.splice(i, 1);
            continue;
          }
          fgCtx.strokeStyle = sl.color;
          fgCtx.globalAlpha = sl.alpha;
          fgCtx.lineWidth = state.isTurbo ? 2.5 : 1.2;
          fgCtx.beginPath();
          fgCtx.moveTo(sl.x, sl.y);
          fgCtx.lineTo(sl.x + sl.len, sl.y);
          fgCtx.stroke();
        }
        fgCtx.restore();

        // B. Trick Confetti & Sparkles
        fgCtx.save();
        for (let i = trickBursts.length - 1; i >= 0; i--) {
          const tb = trickBursts[i];
          tb.x += tb.vx * dt;
          tb.y += tb.vy * dt;
          tb.vy += 220 * dt; // Gravity
          tb.rot += tb.rotV * dt;
          tb.life -= tb.decay * dt;

          if (tb.life <= 0) {
            trickBursts.splice(i, 1);
            continue;
          }

          fgCtx.save();
          fgCtx.translate(tb.x, tb.y);
          fgCtx.rotate(tb.rot);
          fgCtx.globalAlpha = tb.life;
          fgCtx.fillStyle = tb.color;

          // Draw small sparkling diamond / star
          const sz = tb.r * tb.life;
          fgCtx.beginPath();
          fgCtx.moveTo(0, -sz);
          fgCtx.lineTo(sz * 0.6, 0);
          fgCtx.lineTo(0, sz);
          fgCtx.lineTo(-sz * 0.6, 0);
          fgCtx.closePath();
          fgCtx.fill();
          fgCtx.restore();
        }
        fgCtx.restore();

        requestAnimationFrame(animate);
      }

      requestAnimationFrame(animate);
    })();
