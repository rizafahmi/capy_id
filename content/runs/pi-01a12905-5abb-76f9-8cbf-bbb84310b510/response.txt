(() => {

    },
    {
      name: 'Vapor Dream',
      primary: '#ff71ce',
      secondary: '#01cdfe',
      accent: '#05ffa1',
      skyTop: '#100524',
      skyMid1: '#250b4a',
      skyMid2: '#47146e',
      skyMid3: '#79208a',
      skyLow: '#ab329d',
      skyBottom: '#ff9ad9',

      sunMid: '#ff71ce',
      sunBottom: '#2d0854',
      roadBase: '#0e041e',
      gridColor: 'rgba(255, 113, 206, 0.28)'
    }
  ];

  let currentThemeIdx = 0;
  let theme = THEMES[0];

  // CANVAS & VIEWPORT DIMENSIONS
  let width = window.innerWidth;
  let height = window.innerHeight;
  let dpr = window.devicePixelRatio || 1;

  // RIDER STATE
  const state = {
    x: width * 0.44,
    y: height * 0.72,
    baseY: height * 0.72,
    vx: 0,
    vy: 0,
    targetX: width * 0.44,
    targetY: height * 0.72,
    tilt: 0,
    targetTilt: 0,
    // Jump & Trick
    jumping: false,
    jumpTimer: 0,
    jumpDuration: 0.85,
    jumpHeight: 140,
    spinAngle: 0,
    trickName: '',
    // Speed
    turbo: false,
    baseSpeed: 88,
    curSpeed: 88,
    // Customization
    shades: false,
    yuzu: true,
    autoCruise: true,
    autoWanderTimer: 0
  };

  // Keyboard input
  const keys = {
    ArrowLeft: false,
    ArrowRight: false,
    ArrowUp: false,
    ArrowDown: false,
    KeyA: false,
    KeyD: false,
    KeyW: false,
    KeyS: false,
    Space: false,
    ShiftLeft: false,
    ShiftRight: false
  };

  // Particle systems
  const thrusterParticles = [];
  const ambientSparks = [];
  const speedLines = [];

  // Initialize Speed Lines
  for (let i = 0; i < 22; i++) {
    speedLines.push({
      x: Math.random() * width,
      y: Math.random() * height * 0.9,
      len: 70 + Math.random() * 120,
      spd: 12 + Math.random() * 16,
      alpha: 0.2 + Math.random() * 0.5
    });
  }

  // Initialize Ambient sparks
  for (let i = 0; i < 45; i++) {
    ambientSparks.push({
      x: Math.random() * width,
      y: height * 0.35 + Math.random() * height * 0.6,
      rad: 1 + Math.random() * 2.2,
      spd: 1.5 + Math.random() * 3,
      pulse: Math.random() * Math.PI * 2,
      isPrimary: Math.random() > 0.5
    });
  }

  // Window Resize
  function resize() {
    width = window.innerWidth;
    height = window.innerHeight;
    dpr = window.devicePixelRatio || 1;
    canvas.width = width * dpr;
    canvas.height = height * dpr;
    ctx.scale(dpr, dpr);
    state.baseY = height * 0.72;
    if (!state.jumping) {
      state.y = state.baseY;
      state.targetY = state.baseY;
    }
  }

  window.addEventListener('resize', resize);
  resize();

  // Input Listeners
  window.addEventListener('keydown', (e) => {
    if (keys.hasOwnProperty(e.code)) {
      keys[e.code] = true;
      state.autoCruise = false;
      updateAutoCruiseUI();
    }
    if (e.code === 'Space') {
      e.preventDefault();
      triggerTrick();
    }
    if (e.code === 'ShiftLeft' || e.code === 'ShiftRight') {
      state.turbo = true;
    }
    if (e.code === 'KeyS' && !e.ctrlKey && !e.metaKey) {
      toggleShades();
    }
    if (e.code === 'KeyO' || e.code === 'KeyY') {
      toggleYuzu();
    }
    if (e.code === 'KeyC' || e.code === 'KeyT') {
      cycleTheme();
    }
    if (e.code === 'KeyA' && (e.altKey || !keys.KeyA)) {
      state.autoCruise = !state.autoCruise;
      updateAutoCruiseUI();
    }
  });

  window.addEventListener('keyup', (e) => {
    if (keys.hasOwnProperty(e.code)) {
      keys[e.code] = false;
    }
    if (e.code === 'ShiftLeft' || e.code === 'ShiftRight') {
      state.turbo = false;
    }
  });

  // Pointer / Touch Drag
  let isPointerDown = false;
  let lastPointerX = 0;
  let lastPointerY = 0;

  window.addEventListener('pointerdown', (e) => {
    if (e.target.closest('.hud-card') || e.target.closest('.controls-row')) return;
    isPointerDown = true;
    lastPointerX = e.clientX;
    lastPointerY = e.clientY;
    state.autoCruise = false;
    updateAutoCruiseUI();

    // Tap/Click jumps
    triggerTrick();
  });

  window.addEventListener('pointermove', (e) => {
    if (!isPointerDown) return;
    const dx = e.clientX - lastPointerX;
    const dy = e.clientY - lastPointerY;
    lastPointerX = e.clientX;
    lastPointerY = e.clientY;

    const baseScale = Math.min(1.0, Math.max(0.55, width / 680));
    const margin = Math.max(90, 150 * baseScale);
    state.targetX = Math.max(margin, Math.min(width - margin, state.targetX + dx * 1.25));
    state.targetY = Math.max(height * 0.45, Math.min(height * 0.82, state.targetY + dy * 1.25));
  });

  window.addEventListener('pointerup', () => { isPointerDown = false; });
  window.addEventListener('pointercancel', () => { isPointerDown = false; });

  // Buttons
  btnTrick.addEventListener('click', (e) => {
    e.stopPropagation();
    triggerTrick();
  });

  btnTurbo.addEventListener('pointerdown', (e) => {
    e.stopPropagation();
    state.turbo = true;
    btnTurbo.classList.add('active');
  });

  window.addEventListener('pointerup', () => {
    if (state.turbo && !keys.ShiftLeft && !keys.ShiftRight) {
      state.turbo = false;
      btnTurbo.classList.remove('active');
    }
  });

  btnShades.addEventListener('click', (e) => {
    e.stopPropagation();
    toggleShades();
  });

  btnYuzu.addEventListener('click', (e) => {
    e.stopPropagation();
    toggleYuzu();
  });

  btnTheme.addEventListener('click', (e) => {
    e.stopPropagation();
    cycleTheme();
  });

  btnAuto.addEventListener('click', (e) => {
    e.stopPropagation();
    state.autoCruise = !state.autoCruise;
    updateAutoCruiseUI();
  });

  function toggleShades() {
    state.shades = !state.shades;
    btnShades.classList.toggle('active', state.shades);
    showTrickText(state.shades ? 'SHADES EQUIPPED 🕶️' : 'ZEN GAZE 👀');
  }

  function toggleYuzu() {
    state.yuzu = !state.yuzu;
    btnYuzu.classList.toggle('active', state.yuzu);
    showTrickText(state.yuzu ? 'CITRUS BALANCE 🍊' : 'HATLESS MODE 🌿');
  }

  function cycleTheme() {
    currentThemeIdx = (currentThemeIdx + 1) % THEMES.length;
    theme = THEMES[currentThemeIdx];
    themeNameDisp.textContent = theme.name;
    showTrickText(theme.name.toUpperCase() + ' 🎨');
  }

  function updateAutoCruiseUI() {
    autoStateDisp.textContent = state.autoCruise ? 'ON' : 'OFF';
    btnAuto.classList.toggle('active', state.autoCruise);
  }

  const TRICK_NAMES = [
    'SICK KICKFLIP +500',
    '360 HEEL-CHILL +750',
    'ZEN POP-SHOVIT +600',
    'NO DRAMA FLIP +800',
    'MAXIMUM VIBE +1000',
    'OK I PULL UP 🦫✨'
  ];

  function triggerTrick() {
    if (state.jumping) return;
    state.jumping = true;
    state.jumpTimer = 0;
    state.trickName = TRICK_NAMES[Math.floor(Math.random() * TRICK_NAMES.length)];
    showTrickText(state.trickName);
  }

  let trickBannerTimeout = null;
  function showTrickText(text) {
    trickBanner.textContent = text;
    trickBanner.classList.add('show');
    clearTimeout(trickBannerTimeout);
    trickBannerTimeout = setTimeout(() => {
      trickBanner.classList.remove('show');
    }, 1400);
  }

  // Animation Loop Variables
  let lastTime = performance.now();
  let timeSeconds = 0;

  function loop(now) {
    const dt = Math.min(0.05, (now - lastTime) / 1000);
    lastTime = now;
    timeSeconds += dt;

    update(dt);
    render(timeSeconds);

    requestAnimationFrame(loop);
  }

  // Physics & State Updates
  function update(dt) {
    // 1. Auto Cruise Wandering AI
    if (state.autoCruise) {
      state.autoWanderTimer += dt;
      const baseScale = Math.min(1.0, Math.max(0.55, width / 680));
      const margin = Math.max(100, 160 * baseScale);
      const wanderAmp = Math.max(20, (width - margin * 2) * 0.35);
      state.targetX = (width * 0.46) + Math.sin(state.autoWanderTimer * 0.75) * wanderAmp;
      state.targetY = state.baseY + Math.sin(state.autoWanderTimer * 1.4) * 20;

      // Casual automatic trick after initial cruise
      if (state.autoWanderTimer > 5.0 && (state.autoWanderTimer % 10.0 < dt) && !state.jumping) {
        triggerTrick();
      }
    } else {
      // Manual Steering
      const moveSpeed = 380 * dt;
      if (keys.ArrowLeft || keys.KeyA) state.targetX -= moveSpeed;
      if (keys.ArrowRight || keys.KeyD) state.targetX += moveSpeed;
      if (keys.ArrowUp || keys.KeyW) state.targetY -= moveSpeed * 0.7;
      if (keys.ArrowDown || keys.KeyS) state.targetY += moveSpeed * 0.7;

      const baseScale = Math.min(1.0, Math.max(0.55, width / 680));
      const margin = Math.max(90, 150 * baseScale);
      state.targetX = Math.max(margin, Math.min(width - margin, state.targetX));
      state.targetY = Math.max(height * 0.45, Math.min(height * 0.82, state.targetY));
    }

    // Smooth Spring-Mass Position Smoothing
    const springSpeed = 6.0 * dt;
    state.vx = (state.targetX - state.x) * springSpeed;
    state.x += state.vx;

    // Smooth Banking Angle based on horizontal glide velocity
    state.targetTilt = (state.targetX - state.x) * 0.0009;
    state.tilt += (state.targetTilt - state.tilt) * (8.0 * dt);

    // Jump / Ollie / Trick Physics
    let jumpOffset = 0;
    if (state.jumping) {
      state.jumpTimer += dt;
      const progress = state.jumpTimer / state.jumpDuration;

      if (progress >= 1.0) {
        state.jumping = false;
        state.jumpTimer = 0;
        state.spinAngle = 0;
      } else {
        // Jump parabolic arc
        jumpOffset = Math.sin(progress * Math.PI) * state.jumpHeight;
        // 360 Spin
        state.spinAngle = progress * Math.PI * 2;
      }
    }

    // Gentle hoverboard harmonic float oscillation
    const floatBob = Math.sin(timeSeconds * 3.5) * 11 + Math.sin(timeSeconds * 6.7) * 3;
    state.y = (state.targetY - jumpOffset) + floatBob;

    // Turbo Speed Interpolation
    const targetSpeed = state.turbo ? 142 : state.baseSpeed;
    state.curSpeed += (targetSpeed - state.curSpeed) * (5.0 * dt);
    speedDisp.textContent = Math.round(state.curSpeed);

    if (state.turbo) {
      btnTurbo.classList.add('active');
    } else if (!keys.ShiftLeft && !keys.ShiftRight) {
      btnTurbo.classList.remove('active');
    }

    // Spawn Thruster Plasma Particles from Kicktail
    const spawnRate = state.turbo ? 4 : 2;
    for (let i = 0; i < spawnRate; i++) {
      thrusterParticles.push({
        x: state.x - 175 + (Math.random() - 0.5) * 8,
        y: state.y - 6 + (Math.random() - 0.5) * 6,
        vx: -(state.turbo ? 380 : 220) - Math.random() * 160,
        vy: (Math.random() - 0.5) * 45,
        size: 3 + Math.random() * 5,
        alpha: 1.0,
        decay: 1.8 + Math.random() * 2.5,
        color: Math.random() > 0.4 ? theme.primary : theme.secondary
      });
    }

    // Update Particles
    for (let i = thrusterParticles.length - 1; i >= 0; i--) {
      const p = thrusterParticles[i];
      p.x += p.vx * dt;
      p.y += p.vy * dt;
      p.alpha -= p.decay * dt;
      p.size = Math.max(0, p.size - dt * 2.2);
      if (p.alpha <= 0 || p.size <= 0) {
        thrusterParticles.splice(i, 1);
      }
    }

    // Update Speed Lines
    const speedMult = state.turbo ? 2.5 : 1.2;
    speedLines.forEach(l => {
      l.x -= l.spd * speedMult * 60 * dt;
      if (l.x < -200) {
        l.x = width + 50 + Math.random() * 150;
        l.y = Math.random() * height * 0.9;
      }
    });

    // Update Ambient Floating Sparks
    ambientSparks.forEach(s => {
      s.x -= s.spd * speedMult * 35 * dt;
      s.pulse += dt * 3;
      if (s.x < -50) {
        s.x = width + 50 + Math.random() * 80;
        s.y = height * 0.35 + Math.random() * height * 0.6;
      }
    });
  }

    }
    ctx.lineTo(width, horizonY);
    ctx.closePath();
    ctx.fill();

    ctx.strokeStyle = theme.secondary;
    ctx.globalAlpha = 0.35;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.globalAlpha = 1.0;

    ambientSparks.forEach(s => {

    const groundY = Math.min(height - 35, state.baseY + 80);
    ctx.save();
    ctx.translate(state.x, groundY);

    // Contact shadow
    const shadowScale = Math.max(0.4, 1 - ((state.baseY - state.y) / 250));
    const shadowGrad = ctx.createRadialGradient(0, 0, 10, 0, 0, 150 * shadowScale);
    shadowGrad.addColorStop(0, 'rgba(0, 0, 0, 0.85)');
    shadowGrad.addColorStop(0.5, 'rgba(0, 0, 0, 0.4)');
    shadowGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');

    ctx.ellipse(0, 0, 140 * shadowScale, 22 * shadowScale, 0, 0, Math.PI * 2);

    ctx.globalCompositeOperation = 'lighter';
    [-75, 75].forEach(rx => {
      const puddleGrad = ctx.createRadialGradient(rx * shadowScale, 0, 5, rx * shadowScale, 0, 70 * shadowScale);

    ctx.save();
    ctx.globalCompositeOperation = 'lighter';
    thrusterParticles.forEach(p => {
      ctx.fillStyle = p.color;
      ctx.globalAlpha = Math.max(0, p.alpha);

    });
    ctx.restore();

    ctx.save();
    ctx.translate(state.x, state.y);
    ctx.rotate(state.tilt + state.spinAngle);

    drawCapybaraAndBoard(t);

    ctx.restore();
    ctx.restore(); // end camera shake

    ctx.fillStyle = '#060201';
    ctx.beginPath();
    ctx.ellipse(125, 12, 6.5, 3.5, -0.35, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    ctx.arc(124, 10, 3.2, 0.2, Math.PI * 0.9);
    ctx.stroke();

    // Content, serene smile (upturned at the cheek!)
    ctx.strokeStyle = '#281105';
    ctx.lineWidth = 3.2;
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(126, 30);
    ctx.quadraticCurveTo(106, 32, 90, 28);
    ctx.quadraticCurveTo(74, 24, 66, 18);
    ctx.stroke();

    // Whiskers
    ctx.strokeStyle = 'rgba(255, 245, 230, 0.55)';
    ctx.lineWidth = 1.3;
    [
      [106, 18, 146, 20],
      [109, 26, 150, 32],
      [104, 33, 142, 42]
    ].forEach(([x1, y1, x2, y2]) => {
      ctx.beginPath();
      ctx.moveTo(x1, y1);
      ctx.quadraticCurveTo((x1 + x2) / 2, (y1 + y2) / 2 - 2, x2, y2);
      ctx.stroke();
    });

    // Capybara Eye (Zen & sleepy)
    ctx.save();
    ctx.translate(38, -14);

    ctx.fillStyle = '#48240f';
    ctx.beginPath();
    ctx.ellipse(0, 0, 16, 11, -0.15, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = '#1a0b04';
    ctx.beginPath();
    ctx.ellipse(0, 0, 12, 8.5, -0.15, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = '#834213';
    ctx.beginPath();
    ctx.ellipse(1, 2, 7, 4, 0, 0, Math.PI * 2);
    ctx.fill();

    // Natural Eye Blinking animation (brief blink every 3.5s)
    const blinkCycle = (t * 0.28) % 1;
    const isBlinking = blinkCycle > 0.96;
    const lidDrop = isBlinking ? 8.5 : 2.5;

    ctx.fillStyle = '#9a5c2d';
    ctx.beginPath();
    ctx.moveTo(-16, -4);
    ctx.quadraticCurveTo(0, -13, 16, -5);
    ctx.quadraticCurveTo(0, lidDrop, -16, -4);
    ctx.closePath();
    ctx.fill();

    // Specular shine catchlights
    if (!isBlinking) {
      ctx.fillStyle = '#ffffff';
      ctx.beginPath();
      ctx.arc(3.5, -2.5, 2.4, 0, Math.PI * 2);
      ctx.arc(-1, 0, 1.2, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();

    // Capybara Ear (Cute rounded cup on skull)
    ctx.save();
    ctx.translate(-26, -38);
    ctx.rotate(-0.12 + earWiggle);

    ctx.fillStyle = '#54290f';
    ctx.beginPath();
    ctx.ellipse(0, 0, 14, 19, 0.35, 0, Math.PI * 2);
    ctx.fill();

    const earInner = ctx.createLinearGradient(0, -9, 2, 9);
    earInner.addColorStop(0, '#b86654');
    earInner.addColorStop(1, '#df9482');
    ctx.fillStyle = earInner;
    ctx.beginPath();
    ctx.ellipse(2, 2, 8, 12, 0.35, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();

    // RETRO CYBER SUNGLASSES (Toggleable)
    if (state.shades) {
      ctx.save();
      ctx.translate(48, -14);

      ctx.fillStyle = '#0a0d14';
      ctx.strokeStyle = primary;
      ctx.lineWidth = 2.2;
      ctx.shadowColor = primary;
      ctx.shadowBlur = 8;

      ctx.beginPath();
      ctx.moveTo(-34, -14);
      ctx.lineTo(28, -10);
      ctx.quadraticCurveTo(36, -9, 38, -1);
      ctx.lineTo(35, 14);
      ctx.quadraticCurveTo(32, 22, 20, 22);
      ctx.lineTo(-26, 17);
      ctx.quadraticCurveTo(-36, 14, -36, 2);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();

      // Mirror lens
      const lensGrad = ctx.createLinearGradient(-30, -10, 28, 16);
      lensGrad.addColorStop(0, secondary);
      lensGrad.addColorStop(0.5, '#8925d4');
      lensGrad.addColorStop(1, primary);
      ctx.fillStyle = lensGrad;
      ctx.beginPath();
      ctx.roundRect(-30, -8, 60, 24, 5);
      ctx.fill();

      // Moving specular glare
      const gx = -35 + ((t * 1.5) % 4) * 30;
      ctx.save();
      ctx.beginPath();
      ctx.roundRect(-30, -8, 60, 24, 5);
      ctx.clip();
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.85)';
      ctx.lineWidth = 3.5;
      ctx.beginPath();
      ctx.moveTo(gx, 28);
      ctx.lineTo(gx + 18, -18);
      ctx.stroke();
      ctx.restore();

      // Frame temple arm
      ctx.strokeStyle = '#0a0d14';
      ctx.lineWidth = 2.5;
      ctx.shadowBlur = 0;
      ctx.beginPath();
      ctx.moveTo(-34, -8);
      ctx.lineTo(-68, -22);
      ctx.stroke();
      ctx.restore();
    }

    // THE GOLDEN YUZU / ORANGE ON HEAD (Toggleable)
    if (state.yuzu) {
      ctx.save();
      const yuzuWobble = Math.sin(t * 4.5) * 0.1 - headTilt * 0.7;
      ctx.translate(30, -50);
      ctx.rotate(yuzuWobble);

      const yuzuGrad = ctx.createRadialGradient(-5, -6, 3, 0, 0, 20);
      yuzuGrad.addColorStop(0, '#fff47a');
      yuzuGrad.addColorStop(0.35, '#ffb800');
      yuzuGrad.addColorStop(0.8, '#f57c00');
      yuzuGrad.addColorStop(1, '#b84e00');
      ctx.fillStyle = yuzuGrad;
      ctx.shadowColor = 'rgba(255, 170, 0, 0.6)';
      ctx.shadowBlur = 10;
      ctx.beginPath();
      ctx.ellipse(0, 0, 18, 16, 0, 0, Math.PI * 2);
      ctx.fill();

      // Pores texture
      ctx.fillStyle = 'rgba(180, 80, 0, 0.28)';
      ctx.shadowBlur = 0;
      [[-6, -5], [5, -4], [-2, 3], [7, 2], [-8, 2], [3, 7]].forEach(([dx, dy]) => {
        ctx.beginPath();
        ctx.arc(dx, dy, 1, 0, Math.PI * 2);
        ctx.fill();
      });

      // Stem
      ctx.strokeStyle = '#3e6b18';
      ctx.lineWidth = 2.4;
      ctx.beginPath();
      ctx.moveTo(0, -15);
      ctx.quadraticCurveTo(2, -22, 6, -24);
      ctx.stroke();

      // Leaf
      ctx.fillStyle = '#5eb81e';
      ctx.beginPath();
      ctx.moveTo(4, -22);
      ctx.quadraticCurveTo(18, -28, 20, -19);
      ctx.quadraticCurveTo(10, -17, 4, -22);
      ctx.fill();

      ctx.restore();
    }

    ctx.restore(); // end head
    ctx.restore(); // end torso
  }

  // Kick off silky 60fps loop
  requestAnimationFrame(loop);
})();
