// Spawn Collectibles
function spawnCollectible() {
  if (collectibles.length < 5) {
    const lane = (Math.random() - 0.5) * 1.6;
    collectibles.push({
      lane: lane,
      z: 1200 + Math.random() * 400,

      rot: Math.random() * Math.PI * 2,
      bobOffset: Math.random() * 10
    });
  }
}

// Add Floating Popup

    text: text,

    alpha: 1,
    scale: 0.75,
    targetScale: 1.25,

  });
}

// Thruster Ember Particle
function emitThrusterParticles(originX, originY, isBoost) {
  const count = isBoost ? 6 : 3;
  for (let i = 0; i < count; i++) {
    const angle = Math.PI + (Math.random() - 0.5) * 0.5;
    const speed = (isBoost ? 16 : 9) * (0.8 + Math.random() * 0.5);
    particles.push({
      x: originX,
      y: originY + (Math.random() - 0.5) * 12,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed * 0.25 + (Math.random() - 0.5) * 1.8,
      size: (isBoost ? 6 : 4) * (0.6 + Math.random() * 0.6),
      life: 1.0,
      decay: Math.random() * 0.035 + 0.02,
      color: Math.random() > 0.4 ? '#00f0ff' : '#ff007f'
    });
  }
}

// Starburst Fireworks
function emitStarburst(x, y, color = '#ffe600') {
  for (let i = 0; i < 22; i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = Math.random() * 9 + 3;
    particles.push({
      x: x,
      y: y,
      vx: Math.cos(angle) * speed,
      vy: Math.sin(angle) * speed,
      size: Math.random() * 5 + 3,
      life: 1.0,
      decay: Math.random() * 0.04 + 0.02,
      color: color
    });
  }
}

// Ollie / Jump / Trick
function triggerJump() {
  const r = state.rider;
  state.lastUserInput = performance.now();
  if (!r.isJumping) {
    r.isJumping = true;
    r.vy = -16.5;
    r.jumpCount = 1;
    addFloatingText('OLLIE! 🛹', W * 0.5 + r.x * W * 0.35, H * 0.6 + r.yOffset - 60, '#00f0ff');
    emitStarburst(W * 0.5 + r.x * W * 0.35, H * 0.68, '#00f0ff');
  } else if (r.jumpCount === 1) {
    // Air Flip trick!
    r.jumpCount = 2;
    r.vy = -13;
    r.isTricking = true;
    r.spinAngle = 0;
    state.chillLevel = Math.min(100, state.chillLevel + 15);
    addFloatingText('⚡ 360° HOVER-FLIP! +500 ⚡', W * 0.5 + r.x * W * 0.35, H * 0.6 + r.yOffset - 80, '#ffe600');
    emitStarburst(W * 0.5 + r.x * W * 0.35, H * 0.6 + r.yOffset, '#ff007f');
  }
}

// --- Physics & Updates ---
let lastTime = performance.now();

function update(dt) {
  state.time += dt;

  // Auto chill slalom cruise if idle for 2.5s
  if (performance.now() - state.lastUserInput > 2500) {
    state.rider.targetX = Math.sin(state.time * 0.75) * 0.35;
  }

  // Speed and Boost Lerp
  const targetSpeed = state.boost ? 2.3 : 1.0;
  state.speed += (targetSpeed - state.speed) * 0.08;

  // Road grid scrolling
  state.roadZ = (state.roadZ + state.speed * dt * 480) % 200;

  // Rider position lerp
  const r = state.rider;
  r.x += (r.targetX - r.x) * 0.12;

  // Lean / bank angle
  const targetTilt = (r.targetX - r.x) * 0.85;
  r.tilt += (targetTilt - r.tilt) * 0.15;

  // Jump physics
  if (r.isJumping) {
    r.yOffset += r.vy;
    r.vy += 0.88;
    if (r.yOffset >= 0) {
      r.yOffset = 0;
      r.vy = 0;
      r.isJumping = false;
      r.isTricking = false;
      r.spinAngle = 0;
      r.jumpCount = 0;
      emitStarburst(W * 0.5 + r.x * W * 0.35, H * 0.68, '#00f0ff');
    }
  }

  // Trick Flip spin
  if (r.isTricking) {
    r.spinAngle += dt * 14;
    if (r.spinAngle >= Math.PI * 2) {
      r.spinAngle = Math.PI * 2;
    }
  }

  // Spawn collectible
  if (Math.random() < 0.022) {
    spawnCollectible();
  }

  // Roadside Palms scrolling
  for (let palm of roadPalms) {
    palm.z -= state.speed * dt * 480;
    if (palm.z <= 10) {
      palm.z = 1200;
    }
  }

  // Collectibles update & check collision
  const riderScreenX = r.x * (W * 0.35);
  for (let i = collectibles.length - 1; i >= 0; i--) {
    const c = collectibles[i];
    c.z -= state.speed * dt * 520;
    c.rot += dt * 3.2;

    if (c.z < 180 && c.z > 20) {
      const itemScreenX = c.lane * (W * 0.35);
      if (Math.abs(itemScreenX - riderScreenX) < 85 && Math.abs(r.yOffset) < 95) {
        if (c.type === 'yuzu') {
          state.yuzusCollected++;
          state.score += 250;
          state.chillLevel = Math.min(100, state.chillLevel + 10);
          addFloatingText('+250 🍊 JUICY!', W * 0.5 + riderScreenX, H * 0.55 + r.yOffset, '#ffe600');
          emitStarburst(W * 0.5 + riderScreenX, H * 0.55 + r.yOffset, '#ffe600');
        } else {
          state.score += 500;
          state.chillLevel = Math.min(100, state.chillLevel + 20);
          addFloatingText('💎 ULTRA CHILL! +500', W * 0.5 + riderScreenX, H * 0.55 + r.yOffset, '#00f0ff');
          emitStarburst(W * 0.5 + riderScreenX, H * 0.55 + r.yOffset, '#00f0ff');
        }
        collectibles.splice(i, 1);
        continue;
      }
    }

    if (c.z <= 0) {
      collectibles.splice(i, 1);
    }
  }

  // Particles
  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.life -= p.decay;
    if (p.life <= 0) particles.splice(i, 1);
  }

  // Floating text
  for (let i = floatingTexts.length - 1; i >= 0; i--) {
    const ft = floatingTexts[i];
    ft.y += ft.vy;
    ft.alpha -= 0.018;
    if (ft.scale < ft.targetScale) ft.scale += 0.05;
    if (ft.alpha <= 0) floatingTexts.splice(i, 1);
  }

  // Speed streaks
  for (let s of speedStreaks) {
    s.z -= state.speed * dt * 920 * s.speed;
    if (s.z <= 10) {
      s.z = 900;
      s.x = (Math.random() - 0.5) * 2.4;
      s.y = Math.random() * 0.95 + 0.05;
    }
  }

  // HUD sync
  const mph = Math.floor(88 * state.speed);
  document.getElementById('speedVal').innerHTML = `${mph} <span style="font-size:12px;color:#00f0ff;">MPH</span>`;
  document.getElementById('yuzuCount').innerText = state.yuzusCollected;
  document.getElementById('meterFill').style.width = `${state.chillLevel}%`;
  document.getElementById('meterText').innerText = state.chillLevel > 90 ? 'MAX CHILL 🔥' : state.chillLevel > 60 ? 'SERENE 🌊' : 'CRUISIN 🍃';
}

// --- Render Methods ---
function drawEnvironment(theme) {
  const vy = H * state.vanishingYRatio;

  // Sky Gradient
  const sky = ctx.createLinearGradient(0, 0, 0, vy);
  sky.addColorStop(0, theme.skyTop);
  sky.addColorStop(0.3, theme.skyMid1);
  sky.addColorStop(0.65, theme.skyMid2);
  sky.addColorStop(0.85, theme.skyMid3);
  sky.addColorStop(1, theme.skyBottom);
  ctx.fillStyle = sky;
  ctx.fillRect(0, 0, W, vy);

  // Retro Slit Sun
  const sunX = W * 0.5;
  const sunY = vy - H * 0.13;
  const sunR = Math.min(W, H) * 0.19;

  const sunGrad = ctx.createLinearGradient(sunX, sunY - sunR, sunX, sunY + sunR);
  sunGrad.addColorStop(0, theme.sunTop);
  sunGrad.addColorStop(0.5, theme.sunMid);
  sunGrad.addColorStop(1, theme.sunBottom);

  ctx.fillStyle = sunGrad;
  ctx.beginPath();
  ctx.arc(sunX, sunY, sunR, 0, Math.PI * 2);
  ctx.fill();

  // Sun Slits
  ctx.fillStyle = theme.skyTop;
  const slitStart = sunY - sunR * 0.25;
  for (let y = slitStart; y < sunY + sunR; y += 14) {
    const h = ((y - slitStart) / (sunR * 1.25)) * 8 + 2;
    ctx.fillRect(sunX - sunR - 15, y, sunR * 2 + 30, h);
  }

  // Parallax Mountains
  drawParallaxMountains(theme, vy);

  // 3D Highway Road
  drawHighway(theme, vy);

  // Roadside 3D Palms
  drawRoadPalms(theme, vy);
}

function drawParallaxMountains(theme, vy) {
  // Far range
  ctx.fillStyle = theme.mountainFar;
  ctx.beginPath();
  ctx.moveTo(0, vy);
  const steps = 10;
  for (let i = 0; i <= steps; i++) {
    const mx = (W / steps) * i;
    const my = vy - (i % 2 === 1 ? H * 0.15 : H * 0.05);
    ctx.lineTo(mx, my);
  }
  ctx.lineTo(W, vy);
  ctx.closePath();
  ctx.fill();

  ctx.strokeStyle = theme.mountainFarLine;
  ctx.lineWidth = 1.6;
  ctx.stroke();

  // Near range
  ctx.fillStyle = theme.mountainNear;
  ctx.beginPath();
  ctx.moveTo(0, vy);
  for (let i = 0; i <= steps + 2; i++) {
    const mx = (W / (steps + 2)) * i;
    const my = vy - (i % 2 === 0 ? H * 0.09 : H * 0.02);
    ctx.lineTo(mx, my);
  }
  ctx.lineTo(W, vy);
  ctx.closePath();
  ctx.fill();

  ctx.strokeStyle = theme.mountainNearLine;
  ctx.lineWidth = 2;
  ctx.stroke();
}

function drawHighway(theme, vy) {
  const roadHeight = H - vy;

  // Road Surface
  const roadGrad = ctx.createLinearGradient(0, vy, 0, H);
  roadGrad.addColorStop(0, theme.gridBase);
  roadGrad.addColorStop(1, '#1b0028');
  ctx.fillStyle = roadGrad;
  ctx.fillRect(0, vy, W, roadHeight);

  // Radial grid lanes
  const cx = W * 0.5;
  const numRadials = 16;
  for (let i = -numRadials; i <= numRadials; i++) {
    const progress = i / numRadials;
    const startX = cx + progress * (W * 0.12);
    const endX = cx + progress * (W * 1.25);

    ctx.strokeStyle = i === 0 ? '#ff007f' : theme.gridRadials;
    ctx.lineWidth = i === 0 ? 3.2 : 1.3;
    ctx.beginPath();
    ctx.moveTo(startX, vy);
    ctx.lineTo(endX, H);
    ctx.stroke();
  }

  // Horizontal rungs scrolling forward
  const numRungs = 16;
  for (let i = 0; i < numRungs; i++) {
    const norm = ((i * (200 / numRungs) + state.roadZ) % 200) / 200;
    const y = vy + Math.pow(norm, 2.5) * roadHeight;
    const alpha = Math.pow(norm, 1.8);

    ctx.strokeStyle = `rgba(0, 240, 255, ${alpha * 0.85})`;
    ctx.lineWidth = 1 + norm * 3.2;
    ctx.beginPath();
    ctx.moveTo(0, y);
    ctx.lineTo(W, y);
    ctx.stroke();
  }
}

function drawRoadPalms(theme, vy) {
  const cx = W * 0.5;
  const roadHeight = H - vy;

  for (let palm of roadPalms) {
    const norm = Math.max(0.01, Math.min(1.0, 1.0 - palm.z / 1200));
    const y = vy + Math.pow(norm, 2.2) * roadHeight;
    const spread = palm.side * (W * 0.58) * norm;
    const x = cx + spread;
    const scale = norm * 1.6;

    if (scale < 0.1) continue;

    ctx.save();
    ctx.translate(x, y);
    ctx.scale(scale * (palm.side < 0 ? -1 : 1), scale);

    ctx.fillStyle = theme.palmSilhouette;
    // Palm trunk
    ctx.beginPath();
    ctx.moveTo(-5, 0);
    ctx.quadraticCurveTo(15, -60, 20, -110);
    ctx.lineTo(26, -110);
    ctx.quadraticCurveTo(20, -60, 5, 0);
    ctx.closePath();
    ctx.fill();

    // Palm neon outline
    ctx.strokeStyle = theme.neonAccent;
    ctx.lineWidth = 1.2;
    ctx.stroke();

    // Fronds
    ctx.translate(22, -110);
    for (let a = -3; a <= 3; a++) {
      ctx.save();
      ctx.rotate(a * 0.42);
      ctx.fillStyle = theme.palmSilhouette;
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.quadraticCurveTo(28, -22, 54, 5);
      ctx.quadraticCurveTo(22, -6, 0, 0);
      ctx.fill();
      ctx.strokeStyle = theme.neonAccent;
      ctx.lineWidth = 1;
      ctx.stroke();
      ctx.restore();
    }
    ctx.restore();
  }
}

// Draw Collectibles in Perspective

// Draw Speed Streaks

  ctx.save();
  for (let s of speedStreaks) {
    const norm = 1.0 - s.z / 900;

    ctx.stroke();
  }
  ctx.restore();
}

  ctx.ellipse(0, 3, 20, 6, 0, 0, Math.PI * 2);
  ctx.fill();

  const yGrad = ctx.createRadialGradient(-6, -18, 2, 0, -15, 24);
  yGrad.addColorStop(0, '#fff59d');
  yGrad.addColorStop(0.3, '#ffca28');
  yGrad.addColorStop(0.8, '#ff9800');
  yGrad.addColorStop(1, '#e65100');
  ctx.fillStyle = yGrad;
  ctx.beginPath();
  ctx.arc(0, -17, 22, 0, Math.PI * 2);
  ctx.fill();

  // Stem
  ctx.fillStyle = '#4e342e';
  ctx.beginPath();
  ctx.arc(0, -38, 4, 0, Math.PI * 2);
  ctx.fill();
  ctx.fillRect(-2, -45, 4, 8);

  // Leaf
  ctx.save();
  ctx.translate(2, -42);
  ctx.rotate(-0.35 + Math.sin(time * 12) * 0.18);
  ctx.fillStyle = '#43a047';
  ctx.beginPath();
  ctx.ellipse(15, -4, 16, 8, -0.2, 0, Math.PI * 2);
  ctx.fill();
  ctx.strokeStyle = '#2e7d32';
  ctx.lineWidth = 1.4;
  ctx.stroke();
  ctx.restore();

  ctx.restore(); // end Yuzu

  ctx.restore(); // end Head

  // --- FLOWING CYBER SCARF ---
  ctx.save();
  ctx.translate(28, -108);

  // Scarf knot
  ctx.fillStyle = theme.scarfColor;
  ctx.beginPath();
  ctx.roundRect(-16, -7, 40, 26, 10);
  ctx.fill();
  ctx.strokeStyle = theme.scarfTrim;
  ctx.lineWidth = 2.8;
  ctx.stroke();

  // Background ribbon tail
  ctx.fillStyle = '#d4005c';
  ctx.beginPath();
  ctx.moveTo(-14, 6);
  const w1 = Math.sin(time * 14) * 16;
  const w2 = Math.cos(time * 14 + 1.2) * 22;
  const w3 = Math.sin(time * 14 + 2.5) * 28;
  ctx.quadraticCurveTo(-55, 0 + w1, -105, -6 + w2);
  ctx.quadraticCurveTo(-145, -12 + w3, -180, w3);
  ctx.lineTo(-176, w3 + 24);
  ctx.quadraticCurveTo(-135, 8 + w2, -98, 12 + w1);
  ctx.quadraticCurveTo(-50, 18, -12, 24);
  ctx.closePath();
  ctx.fill();

  // Foreground ribbon tail
  ctx.fillStyle = theme.scarfColor;
  ctx.beginPath();
  ctx.moveTo(-11, 2);
  const v1 = Math.cos(time * 14 + 0.5) * 14;
  const v2 = Math.sin(time * 14 + 1.8) * 20;
  const v3 = Math.cos(time * 14 + 3.0) * 26;
  ctx.quadraticCurveTo(-50, -5 + v1, -95, -10 + v2);
  ctx.quadraticCurveTo(-135, -16 + v3, -170, v3 - 6);
  ctx.lineTo(-166, v3 + 16);
  ctx.quadraticCurveTo(-125, 2 + v2, -88, 6 + v1);
  ctx.quadraticCurveTo(-44, 12, -9, 18);
  ctx.closePath();
  ctx.fill();

  // Scarf Gold Trim
  ctx.strokeStyle = theme.scarfTrim;
  ctx.lineWidth = 3.2;
  ctx.beginPath();
  ctx.moveTo(-158, v3 - 4);
  ctx.lineTo(-156, v3 + 14);
  ctx.stroke();

  ctx.restore(); // end scarf

  ctx.restore(); // end body scaling

  ctx.restore(); // end Capybara
}

// Draw Particles
function drawParticles() {
  ctx.save();
  for (let p of particles) {
    ctx.fillStyle = p.color;
    ctx.globalAlpha = Math.max(0, p.life);
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
    ctx.fill();
  }
  ctx.restore();
}

// Draw Floating Score & Text
function drawFloatingTexts() {
  ctx.save();
  ctx.textAlign = 'center';
  ctx.font = '900 22px monospace, sans-serif';
  for (let ft of floatingTexts) {
    ctx.save();
    ctx.translate(ft.x, ft.y);
    ctx.scale(ft.scale, ft.scale);
    ctx.globalAlpha = Math.max(0, ft.alpha);
    ctx.fillStyle = ft.color;
    ctx.shadowColor = ft.color;
    ctx.shadowBlur = 12;
    ctx.fillText(ft.text, 0, 0);
    ctx.restore();
  }
  ctx.restore();
}

// --- Animation Loop ---
function loop(now) {
  const dt = Math.min((now - lastTime) / 1000, 0.1);
  lastTime = now;

  update(dt);

  ctx.save();
  ctx.scale(dpr, dpr);
  ctx.clearRect(0, 0, W, H);

  const theme = THEMES[currentThemeIndex];
  drawEnvironment(theme);
  drawSpeedStreaks();
  drawCollectibles(H * state.vanishingYRatio);
  drawCapybaraScene();
  drawParticles();
  drawFloatingTexts();

  ctx.restore();
  requestAnimationFrame(loop);
}

requestAnimationFrame(loop);
