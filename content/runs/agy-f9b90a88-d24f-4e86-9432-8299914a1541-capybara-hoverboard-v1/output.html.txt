  function drawPalmTree(ctx, x, y, scale, lean, color) {
    ctx.save();
    ctx.translate(x, y);
    ctx.scale(scale, scale);
    ctx.strokeStyle = color;
    ctx.fillStyle = color;
    ctx.lineWidth = 8;
    ctx.lineCap = 'round';

    const topX = 28 * lean * 4;
    const topY = -128;

    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.quadraticCurveTo(topX * 0.3, topY * 0.5, topX, topY);
    ctx.stroke();

    const frondAngles = [-2.5, -1.9, -1.3, -0.5, 0.2, 0.8];
    for (let a of frondAngles) {
      ctx.beginPath();
      ctx.moveTo(topX, topY);
      const fx = topX + Math.cos(a) * 52;
      const fy = topY + Math.sin(a) * 36 + 14;
      ctx.quadraticCurveTo(topX + Math.cos(a) * 26, topY + Math.sin(a) * 18 - 10, fx, fy);
      ctx.lineWidth = 5;
      ctx.stroke();
    }
    ctx.restore();

  function updateAndDrawForeground(dt, theme) {
    fgCtx.clearRect(0, 0, width, height);

    // Smooth pointer interpolation
    state.smoothPointerX += (state.pointerX - state.smoothPointerX) * 0.08;
    state.smoothPointerY += (state.pointerY - state.smoothPointerY) * 0.08;
    state.speed += (state.targetSpeed - state.speed) * 0.06;

    // Update speedometer readout
    const displayKmh = Math.round(42 * state.speed + Math.sin(state.time * 3) * 2);
    speedReadout.textContent = displayKmh;

    // Hover bobbing & pitch
    const hoverFreq = 3.2 * (0.85 + state.speed * 0.15);
    const hoverBobY = Math.sin(state.time * hoverFreq) * 11;
    const hoverPitch = Math.cos(state.time * hoverFreq) * 2.2
                     + state.smoothPointerX * 7
                     + (state.isBoosting ? -4.5 : 0);

    let jumpOffsetY = 0;
    let trickRotation = 0;
    let bodySquashY = 1;

    if (state.isJumping) {
      state.jumpProgress += dt * 1.45;
      if (state.jumpProgress >= 1) {
        state.isJumping = false;
        state.jumpProgress = 0;
        vibeLabel.textContent = state.isBoosting ? 'MAXIMUM CITRUS OVERDRIVE' : 'UNBOTHERED HOVER-CRUISE';
        yuzuStatus.textContent = 'YUZU STABLE';
      } else {
        const p = state.jumpProgress;
        jumpOffsetY = -Math.sin(p * Math.PI) * 135;
        if (state.trickType === 0) {
          trickRotation = p * 360;
        } else {
          trickRotation = -p * 360;
        }
        // Subtle squash at takeoff and landing
        if (p < 0.12) {
          bodySquashY = 0.92;
        } else if (p > 0.88) {
          bodySquashY = 0.94;
        }
      }
    }

    const totalX = state.smoothPointerX * 32;
    const totalY = hoverBobY + jumpOffsetY + state.smoothPointerY * 18;
    const totalRot = hoverPitch + trickRotation;

    masterBoardGroup.setAttribute(
      'transform',
      `translate(${totalX.toFixed(2)}, ${totalY.toFixed(2)}) rotate(${totalRot.toFixed(2)} 300 320)`
    );

    // Subtle secondary knee/body breathing bounce
    const bodyBounceY = Math.sin(state.time * hoverFreq - 0.5) * 3.2;
    capyBodyGroup.setAttribute(
      'transform',
      `translate(0, ${bodyBounceY.toFixed(2)}) scale(1, ${bodySquashY.toFixed(3)})`
    );

    // Subtle head counter-stabilization (Capybara keeps its head remarkably level)
    const headPitch = -hoverPitch * 0.35 + Math.sin(state.time * hoverFreq - 0.9) * 1.4;
    capyHeadGroup.setAttribute(
      'transform',
      `rotate(${headPitch.toFixed(2)} 390 165)`
    );

    // Yuzu Physics on Head
    if (state.isJumping) {
      const yuzuArc = -Math.sin(state.jumpProgress * Math.PI) * 38;
      const yuzuSpin = state.jumpProgress * 360;
      yuzuGroup.setAttribute(
        'transform',
        `translate(384, ${(98 + yuzuArc).toFixed(2)}) rotate(${yuzuSpin.toFixed(1)} 0 -11)`
      );
    } else {
      const yuzuBob = Math.max(0, Math.sin(state.time * hoverFreq - 1.1)) * -4.5;
      const yuzuRock = Math.cos(state.time * hoverFreq) * 4;
      yuzuGroup.setAttribute(
        'transform',
        `translate(384, ${(98 + yuzuBob).toFixed(2)}) rotate(${yuzuRock.toFixed(1)} 0 -4)`
      );
    }

    // Dynamic Wind-Blown Scarf Wave
    const wTime = state.time * (8 + state.speed * 5);
    const sY1 = 204 + Math.sin(wTime) * 12;
    const sY2 = 190 + Math.cos(wTime - 1.2) * 18;
    const sY3 = 194 + Math.sin(wTime - 2.4) * (24 * state.speed);
    const sX3 = 52 - (state.speed - 1) * 24;

    const scarfPath = `M 318,198 Q 225,${sY1.toFixed(1)} 140,${sY2.toFixed(1)} T ${sX3.toFixed(1)},${sY3.toFixed(1)}`;
    scarfTailMain.setAttribute('d', scarfPath);
    scarfTailStripe.setAttribute('d', scarfPath);
    scarfTailShadow.setAttribute(
      'd',
      `M 318,203 Q 225,${(sY1 + 6).toFixed(1)} 140,${(sY2 + 6).toFixed(1)} T ${sX3.toFixed(1)},${(sY3 + 6).toFixed(1)}`
    );

    // Thruster Plume Pulsing
    const thrustPulse = 0.85 + Math.sin(state.time * 28) * 0.18 + (state.speed - 1) * 0.35;
    rearPlumeOuter.setAttribute('rx', (44 * thrustPulse).toFixed(1));
    rearPlumeCore.setAttribute('rx', (27 * thrustPulse).toFixed(1));
    frontPlumeOuter.setAttribute('rx', (44 * thrustPulse).toFixed(1));
    frontPlumeCore.setAttribute('rx', (27 * thrustPulse).toFixed(1));

    // Spawn Exhaust Trail Particles from Hoverboard Rear & Thrusters
    const rigRect = masterBoardGroup.getBoundingClientRect();
    const tailX = rigRect.left + rigRect.width * 0.19;
    const tailY = rigRect.top + rigRect.height * 0.72;

    if (Math.random() < 0.85) {
      particles.push({
        x: tailX + (Math.random() - 0.5) * 14,
        y: tailY + (Math.random() - 0.5) * 12,
        vx: -(4.5 + Math.random() * 4) * state.speed,
        vy: (Math.random() - 0.5) * 1.4,
        size: 6 + Math.random() * 10,
        life: 1,
        decay: 0.022 + Math.random() * 0.015,
        color: Math.random() > 0.45 ? theme.accent1 : theme.accent2,
        type: 'ring'
      });
    }

    // Update & Render Foreground Particles
    for (let i = particles.length - 1; i >= 0; i--) {
      const pt = particles[i];
      pt.x += pt.vx;
      pt.y += pt.vy;
      pt.life -= pt.decay;

      if (pt.life <= 0) {
        particles.splice(i, 1);
        continue;
      }

      fgCtx.save();
      fgCtx.globalAlpha = Math.max(0, pt.life);
      if (pt.type === 'ring') {
        fgCtx.strokeStyle = pt.color;
        fgCtx.lineWidth = 2.2;
        fgCtx.beginPath();
        fgCtx.ellipse(
          pt.x,
          pt.y,
          pt.size * (1.5 - pt.life * 0.5),
          pt.size * 0.45 * (1.5 - pt.life * 0.5),
          0, 0, Math.PI * 2
        );
        fgCtx.stroke();
      } else {
        fgCtx.fillStyle = pt.color;
        fgCtx.beginPath();
        fgCtx.arc(pt.x, pt.y, pt.size * pt.life, 0, Math.PI * 2);
        fgCtx.fill();
      }
      fgCtx.restore();
    }

    // High-Speed Wind Streaks
    fgCtx.save();
    for (let i = 0; i < speedLines.length; i++) {
      const sl = speedLines[i];
      sl.x -= sl.speed * state.speed;
      if (sl.x + sl.len < 0) {
        sl.x = width + Math.random() * 200;
        sl.y = Math.random() * height * 0.88;
      }

      const boostAlpha = state.isBoosting ? sl.alpha * 1.9 : sl.alpha * 0.65;
      const grad = fgCtx.createLinearGradient(sl.x, sl.y, sl.x + sl.len * state.speed, sl.y);
      grad.addColorStop(0, 'rgba(255, 255, 255, 0)');
      grad.addColorStop(0.5, `rgba(0, 243, 255, ${boostAlpha.toFixed(2)})`);
      grad.addColorStop(1, 'rgba(255, 255, 255, 0)');

      fgCtx.strokeStyle = grad;
      fgCtx.lineWidth = state.isBoosting ? 2.2 : 1.2;
      fgCtx.beginPath();
      fgCtx.moveTo(sl.x, sl.y);
      fgCtx.lineTo(sl.x + sl.len * state.speed, sl.y);
      fgCtx.stroke();
    }
    fgCtx.restore();
  }

  let lastTime = performance.now();

  function animate(now) {
    const dt = Math.min((now - lastTime) / 1000, 0.05);
    lastTime = now;

    state.time += dt;
    state.scrollOffset += dt * 240 * state.speed;

    const theme = themes[currentThemeIdx];
    drawBackground(dt, theme);
    updateAndDrawForeground(dt, theme);

    requestAnimationFrame(animate);
  }

  resize();
  initWorld();
  requestAnimationFrame(animate);
})();
