    trickSpinGroup.setAttribute(
      'transform',
      `rotate(${state.trickAngle.toFixed(2)}, 0, -35)`
    );

    // Capybara subtle breathing & knee suspension compression
    const breatheScaleY = 1 + Math.sin(t * 4.2) * 0.016 - (state.jumpVy * 0.00008);
    const breatheScaleX = 1 - Math.sin(t * 4.2) * 0.01;
    capyGroup.setAttribute(
      'transform',
      `scale(${breatheScaleX.toFixed(3)}, ${breatheScaleY.toFixed(3)})`
    );

    // Head secondary bob & chill nod
    const headBobY = -132 + Math.sin(t * 3.1 - 0.5) * 2.8;
    const headNod = Math.sin(t * 2.2) * 1.8 + (state.boost ? -3.5 : 0);
    capyHead.setAttribute(
      'transform',
      `translate(82, ${headBobY.toFixed(2)}) rotate(${headNod.toFixed(2)})`
    );

    // Nostril breathing flare
    const nostrilScale = 1 + Math.sin(t * 4.2) * 0.14 + (state.boost ? 0.18 : 0);
    nostrilGroup.setAttribute(
      'transform',
      `translate(108, -14) scale(${nostrilScale.toFixed(2)})`
    );

    // Natural Blinking & Happy Jump Squint
    state.blinkTimer -= dt;
    if (state.blinkTimer <= 0) {
      state.blinkDuration = 0.15;
      state.blinkTimer = 2.6 + Math.random() * 2.8;
    }
    if (state.blinkDuration > 0) {
      state.blinkDuration -= dt;
    }
    const showClosedEye = state.blinkDuration > 0 || state.isJumping;
    eyeOpen.style.display = showClosedEye ? 'none' : 'block';
    eyeBlink.style.display = showClosedEye ? 'block' : 'none';

    // Ear Twitching
    state.earTwitchTimer -= dt;
    if (state.earTwitchTimer <= 0) {
      state.earTwitchAngle = -20 - Math.random() * 14;
      state.earTwitchTimer = 1.6 + Math.random() * 3.2;
    }
    state.earTwitchAngle *= Math.pow(0.008, dt);
    earNear.setAttribute(
      'transform',
      `translate(-24, -38) rotate(${state.earTwitchAngle.toFixed(1)})`
    );
    earFar.setAttribute(
      'transform',
      `translate(-8, -42) rotate(${(-state.earTwitchAngle * 0.6).toFixed(1)})`
    );

    // Cyber Visor Animation (activates when visor toggled OR during Turbo Boost)
    const targetVisor = (state.visorOn || state.boost) ? 1 : 0;
    state.visorAlpha += (targetVisor - state.visorAlpha) * Math.min(1, dt * 10);
    const visorY = -18 - (1 - state.visorAlpha) * 26;
    cyberVisor.setAttribute('opacity', state.visorAlpha.toFixed(2));
    cyberVisor.setAttribute('transform', `translate(46, ${visorY.toFixed(1)})`);

    // Yuzu Tangerine Spring Physics on Capybara's Head
    const headAccelImpulse = (hoverVy * 0.15);
    const springForce = -state.yuzuOffsetY * 165 - state.yuzuVy * 11;
    state.yuzuVy += (springForce - headAccelImpulse) * dt;
    state.yuzuOffsetY += state.yuzuVy * dt;
    if (state.yuzuOffsetY > 3) {
      state.yuzuOffsetY = 3;
      state.yuzuVy *= -0.45;
    }
    state.yuzuTilt = Math.sin(t * 5.2) * 4 + (state.yuzuOffsetY * 0.6);
    yuzuStackGroup.setAttribute(
      'transform',
      `translate(22, ${(-56 + state.yuzuOffsetY).toFixed(2)}) rotate(${state.yuzuTilt.toFixed(2)})`
    );

    // Dynamic Flowing Scarf Wave
    const waveFreq = t * (state.boost ? 18 : 10);
    const w1 = Math.sin(waveFreq) * (state.boost ? 18 : 10);
    const w2 = Math.cos(waveFreq - 1.3) * (state.boost ? 26 : 15);
    const tailLen = state.boost ? -195 : -162;

    scarfTailMain.setAttribute(
      'd',
      `M 62,-68 C -10,${(-72 + w1).toFixed(1)} -90,${(-48 + w2).toFixed(1)} ${tailLen},${(-56 - w1).toFixed(1)} ` +
      `C ${tailLen + 10},${(-34 - w1).toFixed(1)} ${tailLen + 14},${(-28 - w2).toFixed(1)} ${tailLen + 4},${(-26 - w1).toFixed(1)} ` +
      `C -85,${(-22 + w2).toFixed(1)} -10,${(-42 + w1).toFixed(1)} 60,-42 Z`
    );
    scarfTailStripe.setAttribute(
      'd',
      `M 56,-55 C -15,${(-58 + w1).toFixed(1)} -88,${(-36 + w2).toFixed(1)} ${tailLen + 10},${(-42 - w1).toFixed(1)}`
    );
    scarfTailBack.setAttribute(
      'd',
      `M 55,-72 C -15,${(-82 - w1 * 0.7).toFixed(1)} -85,${(-65 + w2 * 0.8).toFixed(1)} ${tailLen + 18},${(-76 + w1).toFixed(1)} ` +
      `C -90,${(-45 + w2 * 0.6).toFixed(1)} -25,${(-52 - w1 * 0.5).toFixed(1)} 50,-48 Z`
    );

    // Rear Jet Plume Pulse
    const plumeScaleX = (state.boost ? 1.85 : 1.0) + Math.sin(t * 38) * 0.16;
    const plumeScaleY = (state.boost ? 1.35 : 0.95) + Math.cos(t * 44) * 0.12;
    rearJetPlume.setAttribute(
      'transform',
      `translate(-195, 34) scale(${plumeScaleX.toFixed(2)}, ${plumeScaleY.toFixed(2)})`
    );

    // Repulsor Pulsing & Rail Node Animation
    const repulseScale = 1 + Math.sin(t * 28) * 0.18;
    repulsorGlowRear.setAttribute('rx', (30 * repulseScale).toFixed(1));
    repulsorGlowFront.setAttribute('rx', (30 * repulseScale).toFixed(1));

    for (let i = 0; i < railNodes.length; i++) {
      const pulse = ((t * (state.boost ? 4.5 : 2.2) + i * 0.25) % 1);
      const xPos = -155 + pulse * 310;
      const yArc = 6.2 - Math.pow(xPos / 165, 2) * 9.5;
      railNodes[i].setAttribute('cx', xPos.toFixed(1));
      railNodes[i].setAttribute('cy', yArc.toFixed(1));
    }

    // Ground Shadow & Neon Reflection Scaling with Altitude
    const altitudeFactor = Math.max(0.28, 1 - Math.abs(state.y - 455) / 260);
    groundShadowGroup.setAttribute('transform', `translate(${state.x.toFixed(1)}, 585)`);
    groundNeonPool.setAttribute('rx', (245 * altitudeFactor).toFixed(1));
    groundNeonPool.setAttribute('opacity', (0.85 * altitudeFactor).toFixed(2));
    groundDarkCore.setAttribute('rx', (155 * altitudeFactor).toFixed(1));
    groundDarkCore.setAttribute('opacity', (0.68 * altitudeFactor).toFixed(2));

    // Update Telemetry HUD
    const displaySpeed = Math.round(state.speed + Math.sin(t * 3.5) * 2);
    const displayAlt = (1.25 - (state.y - 455) * 0.008).toFixed(2);
    hudSpeed.textContent = `${displaySpeed} KM/H`;
    hudAlt.textContent = `${displayAlt} M`;

    // Spawn Continuous Thruster & Repulsor Particles
    const rad = (pitchAngle * Math.PI) / 180;
    const cosA = Math.cos(rad);
    const sinA = Math.sin(rad);

    // Exhaust particles from rear nozzle (-200, 34) relative to rig
    const nozzleWorldX = state.x + (-200 * cosA - 34 * sinA);
    const nozzleWorldY = state.y + (-200 * sinA + 34 * cosA);

    const spawnCount = state.boost ? 3 : 2;
    for (let i = 0; i < spawnCount; i++) {
      thrusterParticles.push({
        x: nozzleWorldX + (Math.random() - 0.5) * 10,
        y: nozzleWorldY + (Math.random() - 0.5) * 8,
        vx: -(240 + Math.random() * 180) * speedFactor,
        vy: (Math.random() - 0.5) * 45,
        size: (state.boost ? 7 : 5) + Math.random() * 5,
        life: 1,
        decay: 1.8 + Math.random() * 1.1,
        color: Math.random() < 0.55 ? '#00f5d4' : (Math.random() < 0.5 ? '#ff2a85' : '#fee440')
      });
    }

    // Periodic downward anti-gravity rings from front & rear repulsors
    if (Math.random() < (state.boost ? 0.45 : 0.28)) {
      const isFront = Math.random() < 0.5;
      const localX = isFront ? 95 : -95;
      const localY = 50;
      repulsorRings.push({
        x: state.x + (localX * cosA - localY * sinA),
        y: state.y + (localX * sinA + localY * cosA),
        rx: 18,
        ry: 4.5,
        maxRx: 52,
        alpha: 0.72,
        color: isFront ? '#00f5d4' : '#38bdf8'
      });
    }

    // Update Thruster Particles
    for (let i = thrusterParticles.length - 1; i >= 0; i--) {
      const p = thrusterParticles[i];
      p.x += p.vx * dt;
      p.y += p.vy * dt;
      p.life -= p.decay * dt;
      if (p.life <= 0) thrusterParticles.splice(i, 1);
    }

    // Update Repulsor Rings
    for (let i = repulsorRings.length - 1; i >= 0; i--) {
      const r = repulsorRings[i];
      r.y += 95 * dt;
      r.x -= state.speed * 2.2 * dt;
      r.rx += (r.maxRx - r.rx) * 4.5 * dt;
      r.ry = r.rx * 0.22;
      r.alpha -= 1.45 * dt;
      if (r.alpha <= 0) repulsorRings.splice(i, 1);
    }

    // Update Burst Sparks
    for (let i = burstSparks.length - 1; i >= 0; i--) {
      const s = burstSparks[i];
      s.x += s.vx * dt;
      s.y += s.vy * dt;
      s.vy += 380 * dt;
      s.life -= s.decay * dt;
      if (s.life <= 0) burstSparks.splice(i, 1);
    }

    // Update Clouds, Pylons, Petals & Speed Lines
    for (const c of clouds) {
      c.x -= c.speed * speedFactor * dt;
      if (c.x + c.w < -250) c.x = 1480;
    }

    for (const p of pylons) {
      p.x -= state.speed * 5.2 * dt;
      if (p.x < -220) p.x += pylons.length * 220;
    }

    for (const petal of ambientPetals) {
      petal.x -= (state.speed * 4.2 + Math.sin(t * 2 + petal.phase) * 20) * dt;
      petal.y += (petal.vy + Math.cos(t * 2.5 + petal.phase) * 14) * dt;
      petal.rot += petal.vRot * dt;
      if (petal.x < -150) petal.x = 1350;
      if (petal.y > 680) petal.y = -20;
    }

    for (const line of speedLines) {
      line.x -= state.speed * 18 * dt;
      if (line.x + line.len < -200) {
        line.x = 1350 + Math.random() * 250;
        line.y = 80 + Math.random() * 540;
      }
    }
  }

  // Render Background & Parallax World
  function renderBackground() {
    const theme = THEMES[state.themeIdx];
    const t = state.t;

    bgCtx.save();
    bgCtx.setTransform(dpr, 0, 0, dpr, 0, 0);

    // Horizon line in viewport coordinates
    const horizonScreenY = viewOffsetY + 515 * viewScale;

    // 1. Full-screen Sky Gradient
    const skyGrad = bgCtx.createLinearGradient(0, 0, 0, horizonScreenY);
    skyGrad.addColorStop(0, theme.skyTop);
    skyGrad.addColorStop(0.45, theme.skyMid);
    skyGrad.addColorStop(0.82, theme.skyLow);
    skyGrad.addColorStop(1, theme.skyHorizon);
    bgCtx.fillStyle = skyGrad;
    bgCtx.fillRect(0, 0, vw, horizonScreenY + 2);

    // 2. Full-screen Ground Plane
    const groundGrad = bgCtx.createLinearGradient(0, horizonScreenY, 0, vh);
    groundGrad.addColorStop(0, theme.groundTop);
    groundGrad.addColorStop(1, theme.groundBot);
    bgCtx.fillStyle = groundGrad;
    bgCtx.fillRect(0, horizonScreenY, vw, vh - horizonScreenY);

    // Switch to logical 1200x750 coordinate transform matching SVG
    bgCtx.setTransform(
      dpr * viewScale,
      0,
      0,
      dpr * viewScale,
      dpr * viewOffsetX,
      dpr * viewOffsetY
    );

    // 3. Twinkling Stars
    for (const s of stars) {
      const twinkle = 0.35 + 0.65 * Math.sin(t * 2.4 + s.phase);
      bgCtx.fillStyle = `rgba(255, 255, 255, ${twinkle.toFixed(2)})`;
      bgCtx.beginPath();
      bgCtx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      bgCtx.fill();
    }

    // 4. Giant Synthwave Sun / Moon with Horizon Scanlines
    const sunX = 600;
    const sunY = 305;
    const sunR = 148;
    const sunGrad = bgCtx.createLinearGradient(sunX, sunY - sunR, sunX, sunY + sunR);
    sunGrad.addColorStop(0, theme.sunColorTop);
    sunGrad.addColorStop(1, theme.sunColorBot);

    bgCtx.save();
    bgCtx.shadowColor = theme.sunColorBot;
    bgCtx.shadowBlur = 42;
    bgCtx.fillStyle = sunGrad;
    bgCtx.beginPath();
    bgCtx.arc(sunX, sunY, sunR, 0, Math.PI * 2);
    bgCtx.fill();
    bgCtx.restore();

    // Retro horizontal slices in lower half of sun
    bgCtx.fillStyle = theme.skyLow;
    for (let i = 0; i < 6; i++) {
      const sliceY = sunY + 22 + i * 20;
      const sliceH = 3 + i * 1.7;
      bgCtx.fillRect(sunX - sunR - 10, sliceY, (sunR + 10) * 2, sliceH);
    }

    // 5. Drifting Soft Clouds
    for (const c of clouds) {
      bgCtx.fillStyle = 'rgba(255, 255, 255, 0.14)';
      bgCtx.beginPath();
      bgCtx.roundRect(c.x, c.y, c.w, c.h, c.h * 0.5);
      bgCtx.fill();
    }

    // 6. Parallax Mountain Ranges
    drawMountainLayer(bgCtx, state.scrollDist * 0.08, 515, 95, 42, theme.mountainFar);
    drawMountainLayer(bgCtx, state.scrollDist * 0.22, 515, 58, 28, theme.mountainNear);

    // 7. Roadside Neon Pylons / Lantern Pillars
    for (const p of pylons) {
      const topY = 515 - p.height;
      bgCtx.fillStyle = '#0f172a';
      bgCtx.fillRect(p.x - 4, topY, 8, p.height);

      // Glowing neon strip on pylon
      bgCtx.fillStyle = p.type === 0 ? '#00f5d4' : '#ff2a85';
      bgCtx.fillRect(p.x - 1.5, topY + 6, 3, p.height - 16);
    }

    // 8. Perspective Hover Highway & Scrolling Grid
    const horizonY = 515;
    bgCtx.strokeStyle = theme.gridColor;
    bgCtx.lineWidth = 1.6;

    // Horizontal perspective lines
    for (let i = 1; i <= 9; i++) {
      const p = Math.pow(i / 9, 1.75);
      const y = horizonY + p * 260;
      bgCtx.beginPath();
      bgCtx.moveTo(-400, y);
      bgCtx.lineTo(1600, y);
      bgCtx.stroke();
    }

    // Scrolling perspective radial lines
    const gridSpacing = 90;
    const gridShift = (state.scrollDist * 0.85) % gridSpacing;
    for (let x = -600 - gridShift; x <= 1800; x += gridSpacing) {
      const dx = x - 600;
      bgCtx.beginPath();
      bgCtx.moveTo(600 + dx * 0.22, horizonY);
      bgCtx.lineTo(600 + dx * 1.45, 790);
      bgCtx.stroke();
    }

    // Glowing Hover Lane Center Dashes
    const dashPeriod = 180;
    const dashOffset = (state.scrollDist * 1.4) % dashPeriod;
    bgCtx.fillStyle = theme.roadGlow;
    for (let x = -300 - dashOffset; x < 1550; x += dashPeriod) {
      bgCtx.beginPath();
      bgCtx.roundRect(x, 582, 78, 5, 2.5);
      bgCtx.fill();
    }

    // Bright Horizon Divider Beam
    bgCtx.strokeStyle = '#00f5d4';
    bgCtx.lineWidth = 2.5;
    bgCtx.beginPath();
    bgCtx.moveTo(-400, horizonY);
    bgCtx.lineTo(1600, horizonY);
    bgCtx.stroke();

    bgCtx.restore();
  }

  // Render Foreground Particles & Dynamic FX
  function renderFX() {
    const theme = THEMES[state.themeIdx];

    fxCtx.save();
    fxCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
    fxCtx.clearRect(0, 0, vw, vh);

    // Align strictly with SVG 1200x750 coordinate space
    fxCtx.setTransform(
      dpr * viewScale,
      0,
      0,
      dpr * viewScale,
      dpr * viewOffsetX,
      dpr * viewOffsetY
    );

    // 1. Repulsor Anti-Gravity Rings
    for (const r of repulsorRings) {
      fxCtx.strokeStyle = r.color;
      fxCtx.globalAlpha = Math.max(0, r.alpha);
      fxCtx.lineWidth = 2.4;
      fxCtx.beginPath();
      fxCtx.ellipse(r.x, r.y, r.rx, r.ry, 0, 0, Math.PI * 2);
      fxCtx.stroke();
    }

    // 2. Thruster Plasma Exhaust Particles
    for (const p of thrusterParticles) {
      fxCtx.fillStyle = p.color;
      fxCtx.globalAlpha = Math.max(0, p.life * 0.88);
      fxCtx.beginPath();
      fxCtx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
      fxCtx.fill();
    }

    // 3. Trick Burst Sparks
    for (const s of burstSparks) {
      fxCtx.fillStyle = s.color;
      fxCtx.globalAlpha = Math.max(0, s.life);
      fxCtx.beginPath();
      fxCtx.arc(s.x, s.y, s.size * s.life, 0, Math.PI * 2);
      fxCtx.fill();
    }

    // 4. Floating Blossom Petals / Glowing Fireflies
    fxCtx.fillStyle = theme.petalColor;
    for (const petal of ambientPetals) {
      fxCtx.save();
      fxCtx.translate(petal.x, petal.y);
      fxCtx.rotate(petal.rot);
      fxCtx.globalAlpha = 0.72;
      fxCtx.beginPath();
      fxCtx.ellipse(0, 0, petal.size, petal.size * 0.55, 0, 0, Math.PI * 2);
      fxCtx.fill();
      fxCtx.restore();
    }

    // 5. High-Speed Wind Streaks
    const speedAlphaMul = state.boost ? 1.85 : 0.75;
    fxCtx.strokeStyle = '#e0fbfc';
    fxCtx.lineWidth = state.boost ? 2.2 : 1.3;
    for (const line of speedLines) {
      fxCtx.globalAlpha = Math.min(0.75, line.alpha * speedAlphaMul);
      fxCtx.beginPath();
      fxCtx.moveTo(line.x, line.y);
      fxCtx.lineTo(line.x + line.len * (state.boost ? 1.6 : 1), line.y);
      fxCtx.stroke();
    }

    fxCtx.restore();
  }

  // Main Animation Loop
  let lastTime = performance.now();
  function animate(now) {
    const dt = Math.min(0.05, (now - lastTime) / 1000);
    lastTime = now;

    update(dt);
    renderBackground();
    renderFX();

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
})();
