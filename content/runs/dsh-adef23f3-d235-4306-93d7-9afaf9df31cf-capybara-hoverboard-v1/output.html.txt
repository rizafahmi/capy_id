<script>
(function () {
  "use strict";

  var scene = document.getElementById("scene");
  var world = document.getElementById("world");
  var starBox = document.getElementById("stars");
  var fx = document.getElementById("fx");

  function rnd(a, b) { return a + Math.random() * (b - a); }

  /* ---- starfield ---- */
  (function makeStars() {
    var frag = document.createDocumentFragment();
    for (var i = 0; i < 78; i++) {
      var s = document.createElement("i");
      var size = rnd(0.8, 2.4).toFixed(2);
      s.style.left = rnd(0, 100).toFixed(2) + "%";
      s.style.top = rnd(0, 56).toFixed(2) + "%";
      s.style.width = size + "px";
      s.style.height = size + "px";
      s.style.animationDelay = (-rnd(0, 5)).toFixed(2) + "s";
      s.style.animationDuration = rnd(2.4, 5.5).toFixed(2) + "s";
      frag.appendChild(s);
    }
    starBox.appendChild(frag);
  })();

  /* ---- speed streaks ---- */
  function makeStreak() {
    var el = document.createElement("i");
    el.className = "streak";
    el.style.top = rnd(14, 80).toFixed(1) + "%";
    el.style.width = rnd(28, 120).toFixed(0) + "px";
    el.style.height = rnd(1.4, 3).toFixed(1) + "px";
    el.style.setProperty("--o", rnd(0.2, 0.7).toFixed(2));
    el.style.setProperty("--dx", (-scene.clientWidth * rnd(1.05, 1.4)).toFixed(0) + "px");
    el.style.animationDuration = rnd(0.42, 1.05).toFixed(2) + "s";
    fx.appendChild(el);
    el.addEventListener("animationend", function () { el.remove(); });
  }

  /* ---- hoverboard sparks ---- */
  function makeSpark() {
    var el = document.createElement("i");
    el.className = "spark";
    var side = Math.random() < 0.5 ? 40 : 60;
    el.style.left = (side + rnd(-6, 6)).toFixed(1) + "%";
    el.style.top = rnd(81, 88).toFixed(1) + "%";
    var size = rnd(2.5, 6).toFixed(1);
    el.style.width = size + "px";
    el.style.height = size + "px";
    el.style.setProperty("--dx", rnd(-190, -40).toFixed(0) + "px");
    el.style.setProperty("--dy", rnd(30, 150).toFixed(0) + "px");
    el.style.animationDuration = rnd(0.5, 1.15).toFixed(2) + "s";
    fx.appendChild(el);
    el.addEventListener("animationend", function () { el.remove(); });
  }

  var streakTimer = setInterval(function () {
    makeStreak();
    if (scene.classList.contains("boost")) { makeStreak(); makeStreak(); }
  }, 105);

  var sparkTimer = setInterval(function () {
    makeSpark();
    if (Math.random() < 0.45) { makeSpark(); }
  }, 85);

  /* ---- boost on tap / click ---- */
  var boostTimeout = null;
  scene.addEventListener("pointerdown", function () {
    scene.classList.add("boost");
    for (var i = 0; i < 22; i++) {
      (function (n) { setTimeout(makeSpark, n * 14); })(i);
    }
    clearTimeout(boostTimeout);
    boostTimeout = setTimeout(function () {
      scene.classList.remove("boost");
    }, 1500);
  });

  /* ---- gentle pointer parallax ---- */
  var raf = null;
  scene.addEventListener("pointermove", function (e) {
    var r = scene.getBoundingClientRect();
    var nx = (e.clientX - r.left) / r.width - 0.5;
    var ny = (e.clientY - r.top) / r.height - 0.5;
    if (raf) return;
    raf = requestAnimationFrame(function () {
      raf = null;
      world.style.transform =
        "translate(" + (nx * 14).toFixed(2) + "px," + (ny * 9).toFixed(2) + "px)";
    });
  });
  scene.addEventListener("pointerleave", function () {
    world.style.transform = "translate(0,0)";
  });

  void streakTimer; void sparkTimer;
})();
</script>
