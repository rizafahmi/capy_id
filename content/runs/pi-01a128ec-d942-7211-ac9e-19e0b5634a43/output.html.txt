function drawBackground() {
  ctx.clearRect(0, 0, width, height);

  const horizonY = height * 0.65;
  const th = currentTheme;

  // 1. Sky Gradient Background
  const skyGrad = ctx.createLinearGradient(0, 0, 0, horizonY);
  skyGrad.addColorStop(0, th.skyTop);
  skyGrad.addColorStop(0.4, th.skyMid);
  skyGrad.addColorStop(1, th.skyBot);
  ctx.fillStyle = skyGrad;
  ctx.fillRect(0, 0, width, horizonY);

  // 2. Twinkling Stars
  stars.forEach(s => {
    s.alpha += Math.sin(time * 5 + s.x * 20) * 0.01;
    ctx.fillStyle = `rgba(255, 255, 255, ${Math.max(0.15, Math.min(1, s.alpha))})`;
    ctx.beginPath();
    ctx.arc(s.x * width, s.y * height, s.r, 0, Math.PI * 2);
    ctx.fill();
  });

  // 3. Huge Glowing Retrowave Sun
  const sunX = width * 0.5;
  const sunRadius = Math.min(width, height) * 0.22;
  const sunY = horizonY - sunRadius * 0.68;

  // Sun outer atmospheric aura
  const sunAura = ctx.createRadialGradient(sunX, sunY, sunRadius * 0.5, sunX, sunY, sunRadius * 2);
  sunAura.addColorStop(0, th.sunAura);
  sunAura.addColorStop(0.6, 'rgba(112, 0, 255, 0.15)');
  sunAura.addColorStop(1, 'rgba(112, 0, 255, 0)');
  ctx.fillStyle = sunAura;
  ctx.beginPath();
  ctx.arc(sunX, sunY, sunRadius * 2, 0, Math.PI * 2);
  ctx.fill();

  // Sun disc
  const sunGrad = ctx.createLinearGradient(sunX, sunY - sunRadius, sunX, sunY + sunRadius);
  sunGrad.addColorStop(0, th.sun1);
  sunGrad.addColorStop(0.38, th.sun2);
  sunGrad.addColorStop(0.88, th.sun3);
  sunGrad.addColorStop(1, th.skyMid);

  ctx.save();
  ctx.fillStyle = sunGrad;
  ctx.beginPath();
  ctx.arc(sunX, sunY, sunRadius, 0, Math.PI * 2);
  ctx.fill();

  // Retrowave horizontal horizon slices
  ctx.fillStyle = th.skyMid;
  for (let y = sunY + 12; y < sunY + sunRadius; y += 7 + (y - sunY) * 0.14) {
    const sliceH = 2.5 + (y - sunY) * 0.09;
    ctx.fillRect(sunX - sunRadius - 10, y, (sunRadius + 10) * 2, sliceH);
  }
  ctx.restore();

  // 4. Distant Parallax Neon Mountains
  const mtnSpeed = (speed / 80) * 0.6;
  mountainOffset = (mountainOffset + mtnSpeed) % 400;
  ctx.save();
  ctx.fillStyle = th.mountain;
  ctx.strokeStyle = th.mtnStroke;
  ctx.lineWidth = 1.2;
  ctx.beginPath();
  ctx.moveTo(0, horizonY);
  for (let x = 0; x <= width + 80; x += 40) {
    const idx = Math.floor((x + mountainOffset) / 40) % mountainPoints.length;
    const my = horizonY - mountainPoints[idx];
    ctx.lineTo(x, my);
  }
  ctx.lineTo(width, horizonY);
  ctx.closePath();
  ctx.fill();
  ctx.stroke();
  ctx.restore();

  // 5. Parallax Skyline
  const skylineSpeed = (speed / 80) * 1.8;
  skylineOffset = (skylineOffset + skylineSpeed) % (skyline.length * 65);

  ctx.save();
  skyline.forEach(b => {
    let bx = (b.x - skylineOffset) % (skyline.length * 65);
    if (bx < -120) bx += skyline.length * 65;
    const by = horizonY - b.h;

    // Building silhouette with neon edge
    ctx.fillStyle = '#0d0517';
    ctx.fillRect(bx, by, b.w, b.h);
    ctx.strokeStyle = `rgba(${b.hue === 280 ? '160, 32, 240' : '0, 246, 255'}, 0.45)`;
    ctx.lineWidth = 1;
    ctx.strokeRect(bx, by, b.w, b.h);

    if (b.windows) {
      ctx.fillStyle = `rgba(${b.hue === 280 ? '255, 0, 128' : '0, 246, 255'}, 0.35)`;
      for (let wy = by + 12; wy < horizonY - 8; wy += 14) {
        for (let wx = bx + 6; wx < bx + b.w - 6; wx += 9) {
          if (Math.sin(wx * 8 + wy) > 0.2) {
            ctx.fillRect(wx, wy, 3.5, 5);
          }
        }
      }
    }
  });
  ctx.restore();

  // 6. Parallax Midground (Cyber Palms & Lamps)
  const landmarkSpeed = (speed / 80) * 5.5;
  landmarksOffset = (landmarksOffset + landmarkSpeed) % 3600;

  landmarks.forEach(lm => {
    let lx = (lm.x - landmarksOffset) % 3600;
    if (lx < -150) lx += 3600;

    if (lm.type === 'palm') {
      ctx.save();
      ctx.translate(lx, horizonY);
      ctx.strokeStyle = th.palm;
      ctx.lineWidth = 4;
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.quadraticCurveTo(-15, -60, -5, -125);
      ctx.stroke();

      ctx.lineWidth = 2.5;
      for (let f = 0; f < 5; f++) {
        const ang = -Math.PI * 0.2 - (f * Math.PI * 0.15);
        ctx.beginPath();
        ctx.moveTo(-5, -125);
        ctx.quadraticCurveTo(Math.cos(ang) * 40 - 5, -125 + Math.sin(ang) * 40, Math.cos(ang) * 75 - 5, -105 + Math.sin(ang) * 55);
        ctx.stroke();
      }
      ctx.restore();
    } else {
      ctx.save();
      ctx.translate(lx, horizonY);
      ctx.strokeStyle = th.lamp;
      ctx.lineWidth = 3;
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.lineTo(0, -110);
      ctx.lineTo(20, -110);
      ctx.stroke();

      const lampGrad = ctx.createRadialGradient(20, -110, 2, 20, -110, 45);
      lampGrad.addColorStop(0, '#ffffff');
      lampGrad.addColorStop(0.35, th.lamp);
      lampGrad.addColorStop(1, 'rgba(255, 0, 136, 0)');
      ctx.fillStyle = lampGrad;
      ctx.beginPath();
      ctx.arc(20, -110, 45, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }
  });

  // 7. High-speed Neon Perspective Grid Highway
  const roadSpeed = (speed / 80) * 18;
  roadOffset = (roadOffset + roadSpeed) % 80;

  const roadGrad = ctx.createLinearGradient(0, horizonY, 0, height);
  roadGrad.addColorStop(0, th.road1);
  roadGrad.addColorStop(0.35, th.road2);
  roadGrad.addColorStop(1, th.road3);
  ctx.fillStyle = roadGrad;
  ctx.fillRect(0, horizonY, width, height - horizonY);

  // Horizontal Grid Lines
  ctx.lineWidth = 1.5;
  for (let d = 0; d < 1; d += 0.05) {
    const curY = horizonY + Math.pow((d + (roadOffset / 80) * 0.05) % 1, 3) * (height - horizonY);
    const alpha = (curY - horizonY) / (height - horizonY);
    ctx.strokeStyle = th.gridH.replace('0.65', (alpha * 0.7).toFixed(2));
    ctx.beginPath();
    ctx.moveTo(0, curY);
    ctx.lineTo(width, curY);
    ctx.stroke();
  }

  // Perspective Vanishing Lines
  const vpX = width * 0.5;
  for (let x = -width; x <= width * 2; x += 110) {
    ctx.strokeStyle = th.gridV;
    ctx.beginPath();
    ctx.moveTo(vpX, horizonY);
    ctx.lineTo(x, height);
    ctx.stroke();
  }

  // Center High-Speed Dashed Road Strip
  ctx.lineWidth = 7;
  for (let i = 0; i < 15; i++) {
    const p1 = Math.pow((i * 0.08 + (roadOffset / 80) * 0.08) % 1, 2.5);
    const p2 = Math.pow(((i + 0.5) * 0.08 + (roadOffset / 80) * 0.08) % 1, 2.5);
    const y1 = horizonY + p1 * (height - horizonY);
    const y2 = horizonY + p2 * (height - horizonY);
    ctx.strokeStyle = th.centerStrip.replace('0.95', (p1 * 0.95).toFixed(2));
    ctx.beginPath();
    ctx.moveTo(vpX, y1);
    ctx.lineTo(vpX, y2);
    ctx.stroke();
  }

  // 8. Thruster Sparks & Speed Wind Streaks
  particles.forEach(p => {
    if (!p.active) return;
    p.x += p.vx;
    p.y += p.vy;
    p.alpha -= p.spark ? 0.035 : 0.022;
    if (p.alpha <= 0) {
      p.active = false;
      return;
    }
    ctx.fillStyle = p.color;
    ctx.globalAlpha = p.alpha;
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
    ctx.fill();
    ctx.globalAlpha = 1;
  });

  // Wind speed streaks when boosting
  if (boost) {
    for (let i = 0; i < 8; i++) {
      const sy = Math.random() * height;
      const len = 180 + Math.random() * 260;
      const sx = Math.random() * width;
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.45)';
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.moveTo(sx, sy);
      ctx.lineTo(sx - len, sy);
      ctx.stroke();
    }
  }
}

let blinkTimer = 0;
let earTwitchTimer = 0;

function animate(timestamp) {
  time = timestamp * 0.001;

  // Arrow / WASD key control
  if (keys['ArrowLeft'] || keys['KeyA']) targetX = Math.max(-1, targetX - 0.04);
  if (keys['ArrowRight'] || keys['KeyD']) targetX = Math.min(1, targetX + 0.04);
  if (keys['ArrowUp'] || keys['KeyW']) targetY = Math.max(-1, targetY - 0.04);
  if (keys['ArrowDown'] || keys['KeyS']) targetY = Math.min(1, targetY + 0.04);

  // Smooth Speed interpolation
  speed += (targetSpeed - speed) * 0.08;
  speedDisplay.textContent = Math.round(speed);

  // Mouse / Pointer Smooth follow
  currentX += (targetX - currentX) * 0.05;
  currentY += (targetY - currentY) * 0.05;

  // Board Hover Oscillation
  const hoverFreq = boost ? 9 : 5;
  const hoverAmp = boost ? 6 : 12;
  const hoverY = Math.sin(time * hoverFreq) * hoverAmp;
  const hoverPitch = Math.cos(time * hoverFreq * 0.8) * 2.5 + (currentX * 4) + (boost ? 4 : 0);
  const hoverRoll = -currentX * 14;

  // Trick 360 spin animation
  let trickRot = 0;
  let trickLift = 0;
  if (tricking) {
    trickProgress += 0.035;
    trickRot = Math.sin(trickProgress * Math.PI) * 360;
    trickLift = -Math.sin(trickProgress * Math.PI) * 45;
    if (trickProgress >= 1) tricking = false;
  }

  // Apply 3D Transform to Rider Container
  const posX = currentX * 110;
  const posY = currentY * 55 + hoverY + trickLift;
  riderContainer.style.transform = `translate(${posX}px, ${posY}px) rotateZ(${hoverPitch + trickRot}deg) rotateX(${hoverRoll}deg) rotateY(${-currentX * 10}deg)`;

  // Thruster Particle Generation
  const rect = riderContainer.getBoundingClientRect();
  const rearJetX = rect.left + rect.width * 0.31;
  const rearJetY = rect.top + rect.height * 0.86;
  const frontJetX = rect.left + rect.width * 0.69;
  const frontJetY = rect.top + rect.height * 0.84;

  if (Math.random() > (boost ? 0.08 : 0.35)) {
    const pColor = Math.random() > 0.4 ? '#00f6ff' : (Math.random() > 0.5 ? '#ff00aa' : '#ffffff');
    const pSpeed = (speed / 80) * 14 + Math.random() * 6;
    spawnParticle(rearJetX, rearJetY, -pSpeed, (Math.random() - 0.5) * 5, Math.random() * 4 + 2, pColor);
    spawnParticle(frontJetX, frontJetY, -pSpeed, (Math.random() - 0.5) * 5, Math.random() * 4 + 2, pColor);
  }

  // Thruster Plume Size Flicker
  const plumeScale = (boost ? 1.5 : 1.0) + Math.sin(time * 30) * 0.15;
  rearPlume.setAttribute('transform', `scale(${plumeScale}) translate(${280 * (1 - plumeScale) / plumeScale}, ${440 * (1 - plumeScale) / plumeScale})`);
  frontPlume.setAttribute('transform', `scale(${plumeScale}) translate(${620 * (1 - plumeScale) / plumeScale}, ${430 * (1 - plumeScale) / plumeScale})`);

  // Ground shadow reaction to height
  const shadowScale = Math.max(0.4, 1 - (posY / 350));
  groundShadow.setAttribute('rx', 270 * shadowScale);
  groundNeon.setAttribute('rx', 220 * shadowScale);

  // Capybara Head Bob (secondary delayed organic motion)
  const headBob = Math.sin(time * hoverFreq - 0.6) * 3.5;
  headGroup.setAttribute('transform', `translate(0, ${headBob}) rotate(${hoverPitch * 0.25}, 560, 200)`);

  // Ear Twitch
  earTwitchTimer += 0.016;
  let earRot = 0;
  if (earTwitchTimer > 3.0) {
    earRot = Math.sin((earTwitchTimer - 3.0) * 25) * 12;
    if (earTwitchTimer > 3.4) earTwitchTimer = 0;
  }
  earGroup.setAttribute('transform', `translate(488, 156) rotate(${earRot}, 12, 18)`);

  // Eye Blink
  blinkTimer += 0.016;
  if (blinkTimer > 4.0) {
    const blinkPhase = (blinkTimer - 4.0) / 0.25;
    if (blinkPhase <= 1) {
      const lidY = Math.sin(blinkPhase * Math.PI) * 7.5;
      eyelidPath.setAttribute('d', `M -14,${lidY} Q 0,${lidY - 8} 14,${lidY - 1} L 14,3 Q 0,-2 -14,3 Z`);
    } else {
      blinkTimer = 0;
      eyelidPath.setAttribute('d', `M -14,0 Q 0,-8 14,-1 L 14,3 Q 0,-2 -14,3 Z`);
    }
  }

  // Citrus Orange Jiggle & Leaf Flutter
  const orangeJiggle = Math.sin(time * hoverFreq * 1.2) * 1.8;
  citrusFruit.setAttribute('transform', `translate(575, ${134 + headBob + orangeJiggle}) rotate(${-hoverPitch * 0.2}, 0, 0)`);
