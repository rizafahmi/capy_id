      trickToast.classList.remove('show');

  const keys = {};
  window.addEventListener('keydown', e => {
    keys[e.key.toLowerCase()] = true;
    keys[e.code] = true;
    if (e.code === 'Space') {
      e.preventDefault();
      triggerJump();
    }
    if (e.key.toLowerCase() === 'y') btnYuzu.click();
    if (e.key.toLowerCase() === 'g') btnShades.click();
    if (e.key.toLowerCase() === 's' && !keys['keys']) btnScarf.click();
    if (e.key.toLowerCase() === 'c') btnTheme.click();

  window.addEventListener('keyup', e => {
    keys[e.key.toLowerCase()] = false;
    keys[e.code] = false;
  });

  let mouseX = W * 0.45;
  let mouseY = H * 0.62;
  let isPointerDown = false;

  window.addEventListener('pointermove', e => {
    mouseX = e.clientX;
    mouseY = e.clientY;
  });

    if (e.target.closest('.controls-panel') || e.target.closest('.hud-badge')) return;
    isPointerDown = true;
    triggerJump();
  });
  window.addEventListener('pointerup', () => {
    isPointerDown = false;
  });

  btnJump.onclick = () => triggerJump();
  btnBoost.addEventListener('pointerdown', () => { keys['shiftleft'] = true; });
  btnBoost.addEventListener('pointerup', () => { keys['shiftleft'] = false; });

  // Capybara & Board Physics State
  const rider = {
    x: W * 0.45,
    y: H * 0.62,
    targetX: W * 0.45,
    targetY: H * 0.62,
    baseY: H * 0.62,
    altitude: 0,
    altVelocity: 0,
    rotation: 0,
    targetRot: 0,
    spin: 0,
    spinVelocity: 0,
    isJumping: false,
    jumpCount: 0,
    breathPhase: 0,
    blinkTimer: 180,
    isBlinking: false,
    earTwitch: 0,
    earTwitchTimer: 240,
    hoverPhase: 0,
    isBoosting: false,
    squashX: 1,
    squashY: 1,
    scale: 1.55 // Hero size
  };

  function triggerJump() {
    if (rider.jumpCount < 2) {
      rider.altVelocity = rider.jumpCount === 0 ? 17 : 20;
      rider.isJumping = true;
      rider.squashX = 0.92;
      rider.squashY = 1.08;
      rider.jumpCount++;
      if (rider.jumpCount === 1) {
        showToast('HOVER OLLIE! 🛹✨');
      } else {
        rider.spinVelocity = Math.PI * 2 * 0.08;
        showToast('360 KICKFLIP! 🌀🔥');
      }
      spawnJumpBlast();
    }
  }

  // Particle System
  const particles = [];
  function spawnParticle(x, y, vx, vy, color, size, life) {
    particles.push({
      x, y, vx, vy, color, size, baseSize: size, life, maxLife: life
    });
  }

  // Anti-gravity Pulse Ripple Rings
  const repulsorRipples = [];
  function spawnRipple(x, y, color) {
    repulsorRipples.push({
      x, y, radius: 8, maxRadius: 55, alpha: 0.8, color
    });
  }

  function spawnThrusterParticles(x, y, isBoost) {
    const count = isBoost ? 8 : 3;
    for (let i = 0; i < count; i++) {
      const angle = Math.PI * 0.98 + (Math.random() - 0.5) * 0.6;
      const speed = (isBoost ? 12 : 6) + Math.random() * 6;
      const vx = -Math.cos(angle) * speed - globalSpeed * 0.35;
      const vy = Math.sin(angle) * speed + (Math.random() - 0.5) * 3;
      const colors = isBoost
        ? ['#00ffff', '#ffffff', '#ff00aa', '#7df9ff']
        : ['#00e5ff', '#38bdf8', '#818cf8', '#ffffff'];
      const color = colors[Math.floor(Math.random() * colors.length)];
      spawnParticle(x, y, vx, vy, color, (isBoost ? 8 : 4.5) + Math.random() * 4, 25 + Math.random() * 20);
    }
  }

  function spawnJumpBlast() {
    for (let i = 0; i < 36; i++) {
      const angle = Math.random() * Math.PI * 2;
      const spd = 4 + Math.random() * 10;
      spawnParticle(
        rider.x - 30 * rider.scale,
        rider.y + 30 * rider.scale,
        Math.cos(angle) * spd - globalSpeed * 0.4,
        Math.sin(angle) * spd + 2,
        ['#ff007f', '#00ffff', '#ffe600', '#ffffff'][Math.floor(Math.random() * 4)],
        5 + Math.random() * 6,
        35 + Math.random() * 25
      );
    }
  }

  // Scarf Cloth Simulation (Ribbon trailing above back in the wind)
  const scarfSegments = [];
  const NUM_SCARF_SEGS = 16;
  for (let i = 0; i < NUM_SCARF_SEGS; i++) {
    scarfSegments.push({ x: rider.x - 10 - i * 10, y: rider.y - 84 });
  }

  // Stars & Speed Lines
  const stars = [];
  for (let i = 0; i < 140; i++) {
    stars.push({
      x: Math.random() * W,
      y: Math.random() * (H * 0.58),
      size: Math.random() * 2.2 + 0.4,
      alpha: Math.random() * 0.8 + 0.2,

    });

  let globalSpeed = 11;
  let groundGridOffset = 0;
  let skylineOffset = 0;
  let clock = 0;

  // -------------------------------------------------------------
  // BACKGROUND RENDERING
  // -------------------------------------------------------------

  function drawSkyAndSun(theme) {
    const horizonY = H * 0.63;

    // Sky Gradient
    const skyGrad = ctx.createLinearGradient(0, 0, 0, horizonY);
    skyGrad.addColorStop(0, theme.skyTop);
    skyGrad.addColorStop(0.55, theme.skyMid);
    skyGrad.addColorStop(1, theme.skyBottom);
    ctx.fillStyle = skyGrad;
    ctx.fillRect(0, 0, W, horizonY);

    // Stars
    for (const star of stars) {
      star.alpha += Math.sin(clock * star.twinkleSpeed) * 0.012;
      star.alpha = Math.max(0.15, Math.min(0.95, star.alpha));
      ctx.fillStyle = `rgba(255, 255, 255, ${star.alpha})`;
      ctx.beginPath();
      ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
      ctx.fill();
    }

    // Shooting Star occasionally
    if (clock % 340 < 25) {
      const p = (clock % 340) / 25;
      const sx = W * 0.7 - p * 350;
      const sy = H * 0.08 + p * 120;
      const starGrad = ctx.createLinearGradient(sx, sy, sx + 80, sy - 28);
      starGrad.addColorStop(0, '#ffffff');
      starGrad.addColorStop(1, 'transparent');
      ctx.strokeStyle = starGrad;
      ctx.lineWidth = 2.5;
      ctx.beginPath();
      ctx.moveTo(sx, sy);
      ctx.lineTo(sx + 80, sy - 28);
      ctx.stroke();
    }

    // Retro Sliced Sun
    const sunRadius = Math.min(W * 0.17, 150);
    const sunX = W * 0.72;
    const sunY = horizonY - sunRadius * 0.5;

    // Sun outer aura
    const sunAura = ctx.createRadialGradient(sunX, sunY, sunRadius * 0.3, sunX, sunY, sunRadius * 2.2);
    sunAura.addColorStop(0, theme.sunGlow);

    sunAura.addColorStop(1, 'transparent');
    ctx.fillStyle = sunAura;
    ctx.beginPath();
    ctx.arc(sunX, sunY, sunRadius * 2.2, 0, Math.PI * 2);
    ctx.fill();

    // Sliced Sun Body

    ctx.beginPath();
    ctx.arc(sunX, sunY, sunRadius, 0, Math.PI * 2);
    ctx.clip();

    sunGrad.addColorStop(0, theme.sunTop);
    sunGrad.addColorStop(1, theme.sunBottom);
    ctx.fillStyle = sunGrad;
    ctx.fillRect(sunX - sunRadius, sunY - sunRadius, sunRadius * 2, sunRadius * 2);

    // Horizontal retro slice cutouts
    const numSlices = 15;
    for (let i = 0; i < numSlices; i++) {
      const sliceProgress = i / numSlices;
      const sliceY = sunY - sunRadius * 0.15 + sliceProgress * (sunRadius * 1.25);
      const sliceH = 2.5 + sliceProgress * 7;
      ctx.fillStyle = theme.skyMid;
      ctx.fillRect(sunX - sunRadius, sliceY, sunRadius * 2, sliceH);
    }
    ctx.restore();

    // Distant Mountains
    ctx.fillStyle = theme.mountains;
    ctx.strokeStyle = theme.mountainRim;
    ctx.lineWidth = 1.8;
    ctx.beginPath();
    ctx.moveTo(0, horizonY);
    for (let i = 0; i < mountainPoints.length; i++) {
      const pt = mountainPoints[i];
      const my = horizonY - 95 + pt.y;
      ctx.lineTo(pt.x, my);

    ctx.stroke();

      ctx.strokeStyle = 'rgba(255, 0, 128, 0.35)';

      ctx.strokeRect(bx, by, b.w, b.h);

  function drawGroundGrid(theme) {
    const horizonY = H * 0.63;
    const groundH = H - horizonY;

    // Reflective Road Surface
    const groundGrad = ctx.createLinearGradient(0, horizonY, 0, H);
    groundGrad.addColorStop(0, '#0c0418');
    groundGrad.addColorStop(0.35, '#16082c');
    groundGrad.addColorStop(1, '#05010e');
    ctx.fillStyle = groundGrad;
    ctx.fillRect(0, horizonY, W, groundH);

    // Perspective Grid Lines
    ctx.save();
    ctx.strokeStyle = theme.gridColor;
    ctx.shadowColor = theme.gridGlow;
    ctx.shadowBlur = 8;
    ctx.lineWidth = 1.3;

    // Vanishing perspective rays
    const numVanishingLines = 28;
    const centerX = W * 0.5;
    for (let i = -numVanishingLines; i <= numVanishingLines; i++) {
      const progress = i / numVanishingLines;
      const bottomX = centerX + progress * (W * 1.35);
      ctx.beginPath();
      ctx.moveTo(centerX, horizonY);
      ctx.lineTo(bottomX, H);
      ctx.stroke();
    }

    // Horizontal moving stripes (exponential perspective curve)
    groundGridOffset = (groundGridOffset + globalSpeed * 0.055) % 1;
    const numHorizLines = 18;
    for (let i = 0; i < numHorizLines; i++) {
      const p = (i + groundGridOffset) / numHorizLines;
      const depth = Math.pow(p, 2.7);
      const ly = horizonY + depth * groundH;
      ctx.globalAlpha = Math.min(1, depth * 1.6);
      ctx.lineWidth = 0.6 + depth * 2.8;
      ctx.beginPath();
      ctx.moveTo(0, ly);
      ctx.lineTo(W, ly);
      ctx.stroke();
    }
    ctx.restore();
  }

  function drawSpeedLines(theme) {
    ctx.save();
    ctx.strokeStyle = theme.speedColor;
    ctx.lineCap = 'round';
    for (const line of speedLines) {
      line.x -= line.speed * (globalSpeed / 9);
      if (line.x < -line.length) {
        line.x = W + Math.random() * 250;
        line.y = Math.random() * H;
        line.length = 80 + Math.random() * 280;
      }
      ctx.globalAlpha = line.alpha * (globalSpeed / 13);
      ctx.lineWidth = 1.8;
      ctx.beginPath();
      ctx.moveTo(line.x, line.y);
      ctx.lineTo(line.x + line.length, line.y);
      ctx.stroke();
    }
    ctx.restore();
  }

        ctx.moveTo(6, -84);
        for (let i = 0; i < scarfSegments.length; i++) {
          const seg = scarfSegments[i];
          const lx = (seg.x - x) / (S * squashX);
          const ly = ((seg.y - y) / (S * squashY)) + layer * 2;
          ctx.lineTo(lx, ly);
        }
        ctx.stroke();
      }

      ctx.restore();
    }

    ctx.restore();
  }

  // -------------------------------------------------------------
  // DYNAMIC GROUND SHADOW & LIGHT REFLECTION
  // -------------------------------------------------------------
  function drawGroundShadow(x, groundY, altitude, tilt, S) {
    ctx.save();
    const shadowDist = Math.max(0, groundY - (rider.y + 20 * S));
    const shadowScale = Math.max(0.35, 1 - shadowDist / 320) * S;
    const shadowAlpha = Math.max(0.12, 0.7 - shadowDist / 340);

    ctx.translate(x, groundY);
    ctx.scale(shadowScale, shadowScale * 0.45);

    // Deep ambient shadow
    const shadowGrad = ctx.createRadialGradient(0, 0, 10, 0, 0, 95);
    shadowGrad.addColorStop(0, `rgba(0, 0, 0, ${shadowAlpha})`);
    shadowGrad.addColorStop(0.55, `rgba(18, 0, 35, ${shadowAlpha * 0.7})`);
    shadowGrad.addColorStop(1, 'transparent');

    ctx.fillStyle = shadowGrad;
    ctx.beginPath();
    ctx.ellipse(0, 0, 100, 38, tilt * 0.4, 0, Math.PI * 2);
    ctx.fill();

    // Wet Road Neon Underglow Reflection
    const glowGrad = ctx.createRadialGradient(0, 0, 5, 0, 0, 75);
    glowGrad.addColorStop(0, `rgba(0, 255, 255, ${shadowAlpha * 0.55})`);
    glowGrad.addColorStop(0.65, `rgba(255, 0, 128, ${shadowAlpha * 0.3})`);
    glowGrad.addColorStop(1, 'transparent');

    ctx.fillStyle = glowGrad;
    ctx.beginPath();
    ctx.ellipse(0, 0, 80, 28, 0, 0, Math.PI * 2);
    ctx.fill();

    ctx.restore();
  }

  // -------------------------------------------------------------
  // MAIN ANIMATION UPDATE & LOOP
  // -------------------------------------------------------------

  function update() {
    clock++;

    // Responsive scaling
    rider.scale = Math.max(1.15, Math.min(1.7, Math.min(W, H) / 500));

    // Boosting
    rider.isBoosting = keys['shift'] || keys['shiftleft'] || keys['shiftright'] || isPointerDown;
    const targetGlobalSpeed = rider.isBoosting ? 28 : 11;
    globalSpeed += (targetGlobalSpeed - globalSpeed) * 0.08;

    // Movement Controls
    if (keys['arrowleft'] || keys['keya']) mouseX -= 9;
    if (keys['arrowright'] || keys['keyd']) mouseX += 9;
    if (keys['arrowup'] || keys['keyw']) mouseY -= 9;
    if (keys['arrowdown'] || keys['keys']) mouseY += 9;

    mouseX = Math.max(W * 0.15, Math.min(W * 0.82, mouseX));
    mouseY = Math.max(H * 0.38, Math.min(H * 0.78, mouseY));

    rider.targetX = mouseX;
    rider.baseY += (mouseY - rider.baseY) * 0.07;
    rider.x += (rider.targetX - rider.x) * 0.08;

    // Hover floating wave
    rider.hoverPhase += 0.052;
    const hoverBob = Math.sin(rider.hoverPhase) * (10 * (rider.scale / 1.5));

    // Jump Physics & Squash/Stretch
    if (rider.isJumping) {
      rider.altitude += rider.altVelocity;
      rider.altVelocity -= 0.88; // Gravity

      if (rider.altitude <= 0) {
        rider.altitude = 0;
        rider.altVelocity = 0;
        rider.isJumping = false;
        rider.jumpCount = 0;
        // Landing impact squash
        rider.squashX = 1.15;
        rider.squashY = 0.86;
        spawnJumpBlast();
      }
    }

    // Squash damping back to 1.0
    rider.squashX += (1 - rider.squashX) * 0.12;
    rider.squashY += (1 - rider.squashY) * 0.12;

    // 360 Spin Trick Physics
    if (rider.spinVelocity > 0) {
      rider.spin += rider.spinVelocity;
      if (rider.spin >= Math.PI * 2) {
        rider.spin = 0;
        rider.spinVelocity = 0;
      }
    }

    // Final Y position
    rider.y = rider.baseY - rider.altitude - hoverBob;

    // Tilt angle
    const moveDiffX = rider.targetX - rider.x;
    rider.targetRot = (moveDiffX * 0.0028) + (rider.altVelocity * -0.015);
    rider.rotation += (rider.targetRot - rider.rotation) * 0.1;

    // Breathing & Idle Twitch timers
    rider.breathPhase += 0.042;
    rider.blinkTimer--;
    if (rider.blinkTimer <= 0) {
      rider.isBlinking = !rider.isBlinking;
      rider.blinkTimer = rider.isBlinking ? 8 : (140 + Math.random() * 180);
    }

    rider.earTwitchTimer--;
    if (rider.earTwitchTimer <= 0) {
      rider.earTwitch += 0.3;
      if (rider.earTwitch > Math.PI * 4) {
        rider.earTwitch = 0;
        rider.earTwitchTimer = 190 + Math.random() * 240;
      }
    }

    // Scarf Physics Simulation (Trailing above back into the wind)
    const scarfAnchorX = rider.x + 6 * rider.scale;
    const scarfAnchorY = rider.y - 84 * rider.scale;
    scarfSegments[0].x = scarfAnchorX;
    scarfSegments[0].y = scarfAnchorY;

    for (let i = 1; i < scarfSegments.length; i++) {
      const seg = scarfSegments[i];
      const prev = scarfSegments[i - 1];
      const windPush = globalSpeed * 1.25;
      const flutter = Math.sin(clock * 0.36 + i * 0.65) * 4.2;

      const targetSegY = prev.y - 1.2 + flutter;
      seg.x += (prev.x - 9.5 * rider.scale - windPush * 0.24 - seg.x) * 0.36;
      seg.y += (targetSegY - seg.y) * 0.36;
    }

    // Periodic repulsor ripple rings
    if (clock % (rider.isBoosting ? 6 : 14) === 0) {
      const groundRefY = rider.baseY + 36 * rider.scale;
      spawnRipple(rider.x - 40 * rider.scale, groundRefY, THEMES[currentThemeIdx].boardGlow);
      spawnRipple(rider.x + 40 * rider.scale, groundRefY, THEMES[currentThemeIdx].boardGlow);
    }

    // Update Ripples
    for (let i = repulsorRipples.length - 1; i >= 0; i--) {
      const rip = repulsorRipples[i];
      rip.radius += 2.5;
      rip.alpha -= 0.035;
      if (rip.alpha <= 0 || rip.radius >= rip.maxRadius) {
        repulsorRipples.splice(i, 1);
      }
    }

    // Thruster exhaust particles
    spawnThrusterParticles(
      rider.x - 85 * rider.scale,
      rider.y + 15 * rider.scale,
      rider.isBoosting
    );

    // Update Particles
    for (let i = particles.length - 1; i >= 0; i--) {
      const p = particles[i];
      p.x += p.vx;
      p.y += p.vy;
      p.life--;
      p.size = p.baseSize * (p.life / p.maxLife);
      if (p.life <= 0) {
        particles.splice(i, 1);
      }
    }

    // Update HUD Stats
    const currentSpeedVal = Math.round(globalSpeed * 12.5);
    statSpd.textContent = currentSpeedVal;
    statAlt.textContent = (1.4 + (rider.altitude / 30)).toFixed(1);
  }

  function draw() {
    ctx.clearRect(0, 0, W, H);
    const theme = THEMES[currentThemeIdx];

    // Background Layers
    drawSkyAndSun(theme);
    drawGroundGrid(theme);
    drawSpeedLines(theme);

    // Dynamic Ground Shadow & Repulsor Shockwave Ripples
    const groundRefY = rider.baseY + 36 * rider.scale;
    drawGroundShadow(rider.x, groundRefY, rider.altitude, rider.rotation, rider.scale);

    // Anti-gravity Expanding Shockwave Ripples on Ground
    ctx.save();
    for (const rip of repulsorRipples) {
      ctx.strokeStyle = rip.color;
      ctx.globalAlpha = rip.alpha;
      ctx.lineWidth = 2;
      ctx.shadowColor = rip.color;
      ctx.shadowBlur = 8;
      ctx.beginPath();
      ctx.ellipse(rip.x, rip.y, rip.radius * 1.6, rip.radius * 0.6, 0, 0, Math.PI * 2);
      ctx.stroke();
    }
    ctx.restore();

    // Thruster Particle Glow
    ctx.save();
    for (const p of particles) {
      ctx.globalAlpha = p.life / p.maxLife;
      ctx.fillStyle = p.color;
      ctx.shadowColor = p.color;
      ctx.shadowBlur = 10;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();

    // Hoverboard and Capybara Hero
    drawHoverboard(rider.x, rider.y + 18 * rider.scale, rider.rotation, theme, rider.isBoosting, rider.scale, rider.squashX, rider.squashY);
    drawCapybara(rider.x, rider.y + 14 * rider.scale, rider.rotation, rider, rider.scale, rider.squashX, rider.squashY);
  }

  function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
  }

  requestAnimationFrame(loop);
})();
