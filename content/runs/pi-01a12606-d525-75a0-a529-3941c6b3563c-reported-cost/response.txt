  ];
  let currentThemeIdx = 0;

  themeBtn.addEventListener('click', (e) => {
    e.stopPropagation();
    currentThemeIdx = (currentThemeIdx + 1) % themes.length;
    themeBtn.textContent = themes[(currentThemeIdx + 1) % themes.length].name + ' MODE';
  });

  // Physics & Rider State
  const state = {
    targetX: width * 0.45,
    targetY: height * 0.52,
    x: width * 0.45,
    y: height * 0.52,
    vx: 0,
    vy: 0,
    tilt: 0,
    targetTilt: 0,
    isJumping: false,
    jumpTimer: 0,
    jumpOffsetY: 0,
    jumpRotation: 0,
    trickType: 0, // 0: ollie, 1: 360 spin, 2: kickflip roll
    turbo: false,
    speed: 88,
    targetSpeed: 88,
    gridOffset: 0
  };

  // Particles
  const thrusterParticles = [];
  const roadParticles = [];
  const speedLines = [];
  const stars = [];

  // Init Stars
  for (let i = 0; i < 90; i++) {
    stars.push({
      x: Math.random() * 2000,
      y: Math.random() * 400,
      size: Math.random() * 1.8 + 0.5,
      alpha: Math.random() * 0.8 + 0.2,
      twinkleSpeed: Math.random() * 0.03 + 0.01
    });
  }

  // Skyline Buildings (procedural)
  const skyline = [];
  let curX = 0;
  while (curX < 4000) {
    const bWidth = Math.random() * 60 + 40;
    const bHeight = Math.random() * 180 + 70;
    const windowCols = Math.floor(bWidth / 12);
    const windowRows = Math.floor(bHeight / 16);
    const windows = [];
    for (let r = 0; r < windowRows; r++) {
      for (let c = 0; c < windowCols; c++) {
        if (Math.random() > 0.4) {
          windows.push({ x: c * 10 + 6, y: r * 14 + 10 });
        }
      }
    }
    skyline.push({
      x: curX,
      width: bWidth,
      height: bHeight,
      windows: windows,
      hasSpire: Math.random() > 0.6
    });
    curX += bWidth + Math.random() * 15;
  }

  // Mouse & Touch Interaction
  window.addEventListener('pointermove', (e) => {
    state.targetX = Math.max(width * 0.15, Math.min(width * 0.85, e.clientX));
    state.targetY = Math.max(height * 0.35, Math.min(height * 0.68, e.clientY));
  });

  window.addEventListener('keydown', (e) => {
    if (e.code === 'Space') {
      triggerTrick();
      e.preventDefault();
    }
    if (e.code === 'KeyW' || e.code === 'ShiftLeft' || e.code === 'ShiftRight') {
      state.turbo = true;
    }
  });

  window.addEventListener('keyup', (e) => {
    if (e.code === 'KeyW' || e.code === 'ShiftLeft' || e.code === 'ShiftRight') {
      state.turbo = false;
    }
  });

  document.getElementById('viewport').addEventListener('click', (e) => {
    if (e.target.closest('.controls-hint')) return;
    triggerTrick();
  });

  trickBtn.addEventListener('click', (e) => {
    e.stopPropagation();
    triggerTrick();
  });

  function triggerTrick() {
    if (state.isJumping) return;
    state.isJumping = true;
    state.jumpTimer = 0;
    state.trickType = Math.floor(Math.random() * 3);
  }

  // Add Thruster Sparks & Dust Rings
  function spawnThrusterParticles() {
    const rad = (state.tilt + state.jumpRotation) * Math.PI / 180;
    const cosT = Math.cos(rad);
    const sinT = Math.sin(rad);
    // Thruster exhaust point relative to 230, 230 center is (-145, 170)
    const flameX = state.x + (-145 * cosT - 170 * sinT);
    const flameY = state.y + (-145 * sinT + 170 * cosT) + state.jumpOffsetY;

    const count = state.turbo ? 6 : 3;
    for (let i = 0; i < count; i++) {
      const spread = (Math.random() - 0.5) * 6;
      thrusterParticles.push({
        x: flameX + (Math.random() - 0.5) * 10,
        y: flameY + (Math.random() - 0.5) * 10,
        vx: -(Math.random() * 8 + (state.turbo ? 16 : 8)) * cosT + spread,
        vy: (Math.random() - 0.5) * 4 - sinT * 6,
        size: Math.random() * 6 + 3,
        color: Math.random() > 0.4 ? '#00f3ff' : '#ff00aa',
        alpha: 1,
        life: 1
      });
    }

    // Ground hover dust / repulsor rings
    const groundY = state.y + 195;
    if (!state.isJumping && Math.random() > 0.55) {
      roadParticles.push({
        x: state.x + (Math.random() - 0.5) * 100,
        y: groundY,
        rx: 12,
        ry: 4.5,
        maxRadius: Math.random() * 35 + 25,
        alpha: 0.7,
        color: Math.random() > 0.5 ? '#00f3ff' : '#ff00aa'
      });
    }

    // Speed lines
    if (Math.random() > (state.turbo ? 0.15 : 0.55)) {
      speedLines.push({
        x: width + 50,
        y: Math.random() * height,
        length: Math.random() * 140 + 70,
        speed: (Math.random() * 22 + 25) * (state.turbo ? 1.8 : 1),
        alpha: Math.random() * 0.45 + 0.25
      });
    }
  }

  // Draw Background Scene
  function drawBackground(theme) {
    const horizon = height * 0.60;

    // Sky gradient
    const skyGrad = bgCtx.createLinearGradient(0, 0, 0, horizon);
    skyGrad.addColorStop(0, theme.skyTop);
    skyGrad.addColorStop(0.55, theme.skyMid);
    skyGrad.addColorStop(1, theme.skyBot);
    bgCtx.fillStyle = skyGrad;
    bgCtx.fillRect(0, 0, width, horizon);

    // Stars
    bgCtx.save();
    for (let s of stars) {
      s.alpha += Math.sin(Date.now() * s.twinkleSpeed) * 0.01;
      bgCtx.fillStyle = `rgba(255, 255, 255, ${Math.max(0.1, Math.min(0.9, s.alpha))})`;
      bgCtx.beginPath();
      bgCtx.arc(s.x % width, s.y % horizon, s.size, 0, Math.PI * 2);
      bgCtx.fill();
    }
    bgCtx.restore();

    // Giant Synthwave Retro Sun
    const sunX = width * 0.75;
    const sunY = horizon - 50;
    const sunR = Math.min(width, height) * 0.18;
    const sunGrad = bgCtx.createLinearGradient(sunX, sunY - sunR, sunX, sunY + sunR);
    sunGrad.addColorStop(0, theme.sunTop);
    sunGrad.addColorStop(1, theme.sunBot);

    bgCtx.save();
    bgCtx.shadowColor = theme.sunBot;
    bgCtx.shadowBlur = 40;
    bgCtx.fillStyle = sunGrad;
    bgCtx.beginPath();
    bgCtx.arc(sunX, sunY, sunR, 0, Math.PI * 2);
    bgCtx.fill();
    bgCtx.restore();

    // Sun horizontal scanline cuts
    bgCtx.save();
    bgCtx.fillStyle = theme.skyBot;
    const cutCount = 7;
    for (let i = 0; i < cutCount; i++) {
      const cutH = (i + 1) * 2.2;
      const cutY = sunY + sunR * 0.15 + (i * 12);
      if (cutY < horizon) {
        bgCtx.fillRect(sunX - sunR - 10, cutY, sunR * 2 + 20, cutH);
      }
    }
    bgCtx.restore();

    // Distant Mountain Ridges
    bgCtx.fillStyle = theme.mountains;
    bgCtx.beginPath();
    bgCtx.moveTo(0, horizon);
    const mtnScroll = (Date.now() * 0.02) % 400;
    for (let x = 0; x <= width + 50; x += 40) {
      const nx = x + mtnScroll;
      const my = horizon - 45 - Math.sin(nx * 0.008) * 35 - Math.sin(nx * 0.02) * 20;
      bgCtx.lineTo(x, my);
    }
    bgCtx.lineTo(width, horizon);
    bgCtx.closePath();
    bgCtx.fill();

    // Parallax Skyline
    const cityScroll = (Date.now() * 0.08) % 2000;
    bgCtx.save();
    bgCtx.fillStyle = theme.buildings;
    for (let b of skyline) {
      const drawX = ((b.x - cityScroll) % 3000 + 3000) % 3000 - 300;
      if (drawX > -150 && drawX < width + 150) {
        bgCtx.fillRect(drawX, horizon - b.height, b.width, b.height);

        // Windows
        bgCtx.fillStyle = theme.sunTop;
        for (let w of b.windows) {
          if ((drawX + w.x) % 7 > 2) {
            bgCtx.fillRect(drawX + w.x, horizon - b.height + w.y, 3, 5);
          }
        }
        bgCtx.fillStyle = theme.buildings;

        // Antennas / spires with flashing light
        if (b.hasSpire) {
          bgCtx.fillRect(drawX + b.width / 2 - 1, horizon - b.height - 20, 2, 20);
          bgCtx.fillStyle = Math.sin(Date.now() * 0.006 + drawX) > 0 ? '#ff0055' : '#00f3ff';
          bgCtx.fillRect(drawX + b.width / 2 - 2, horizon - b.height - 22, 4, 4);
          bgCtx.fillStyle = theme.buildings;
        }
      }
    }
    bgCtx.restore();

    // Lower Ground Grid (Perspective 3D Road)
    const groundGrad = bgCtx.createLinearGradient(0, horizon, 0, height);
    groundGrad.addColorStop(0, theme.groundGrad[0]);
    groundGrad.addColorStop(1, theme.groundGrad[1]);
    bgCtx.fillStyle = groundGrad;
    bgCtx.fillRect(0, horizon, width, height - horizon);

    // Horizon neon glow strip
    bgCtx.save();
    bgCtx.strokeStyle = theme.gridGlow;
    bgCtx.shadowColor = theme.gridGlow;
    bgCtx.shadowBlur = 20;
    bgCtx.lineWidth = 3;
    bgCtx.beginPath();
    bgCtx.moveTo(0, horizon);
    bgCtx.lineTo(width, horizon);
    bgCtx.stroke();
    bgCtx.restore();

    // Grid Perspective Vertical Lines
    bgCtx.save();
    bgCtx.strokeStyle = theme.gridColor;
    bgCtx.lineWidth = 1.2;
    const vanishX = width * 0.5;
    const vanishY = horizon;
    const numGridLines = 26;

    for (let i = -numGridLines; i <= numGridLines; i++) {
      const bottomX = vanishX + (i * (width / numGridLines) * 2.2);
      bgCtx.beginPath();
      bgCtx.moveTo(vanishX, vanishY);
      bgCtx.lineTo(bottomX, height);
      bgCtx.stroke();
    }

    // Grid Perspective Horizontal Lines (scrolling towards screen)
    state.gridOffset = (state.gridOffset + (state.turbo ? 0.024 : 0.012)) % 1;
    const horizLines = 14;
    for (let j = 0; j < horizLines; j++) {
      const p = (j + state.gridOffset) / horizLines;
      const y = horizon + Math.pow(p, 2.5) * (height - horizon);
      bgCtx.strokeStyle = `rgba(0, 243, 255, ${p * 0.6})`;
      bgCtx.lineWidth = 1 + p * 2.5;
      bgCtx.beginPath();
      bgCtx.moveTo(0, y);
      bgCtx.lineTo(width, y);
      bgCtx.stroke();
    }
    bgCtx.restore();
  }

  // Foreground Canvas: Particles, Hoverboard Ground Reflection, Speed Streaks
  function drawForeground() {
    fgCtx.clearRect(0, 0, width, height);

    // Dynamic Under-Board Shadow & Hover Glow on Ground
    const groundY = state.y + 195;
    const jumpDist = Math.abs(state.jumpOffsetY);
    const shadowScale = Math.max(0.25, 1 - jumpDist / 200);
    
    // Board Repulsor Glow pool on asphalt
    const glowGrad = fgCtx.createRadialGradient(
      state.x, groundY, 10,
      state.x, groundY, 150 * shadowScale
    );
    glowGrad.addColorStop(0, `rgba(0, 243, 255, ${0.5 * shadowScale})`);
    glowGrad.addColorStop(0.4, `rgba(255, 0, 170, ${0.25 * shadowScale})`);
    glowGrad.addColorStop(1, 'rgba(0,0,0,0)');
    
    fgCtx.save();
    fgCtx.fillStyle = glowGrad;
    fgCtx.beginPath();
    fgCtx.ellipse(state.x, groundY, 170 * shadowScale, 34 * shadowScale, 0, 0, Math.PI * 2);
    fgCtx.fill();

    // Dark contact shadow beneath board
    fgCtx.fillStyle = `rgba(0, 0, 0, ${0.5 * shadowScale})`;
    fgCtx.beginPath();
    fgCtx.ellipse(state.x, groundY + 2, 110 * shadowScale, 18 * shadowScale, 0, 0, Math.PI * 2);
    fgCtx.fill();
    fgCtx.restore();

    // Repulsor Dust Rings
    for (let i = roadParticles.length - 1; i >= 0; i--) {
      const p = roadParticles[i];
      p.rx += 1.8;
      p.ry += 0.6;
      p.x -= state.turbo ? 9 : 5;
      p.alpha -= 0.025;

      if (p.alpha <= 0) {
        roadParticles.splice(i, 1);
        continue;
      }

      fgCtx.save();
      fgCtx.strokeStyle = p.color;
      fgCtx.lineWidth = 2.5;
      fgCtx.globalAlpha = p.alpha;
      fgCtx.beginPath();
      fgCtx.ellipse(p.x, p.y, p.rx, p.ry, 0, 0, Math.PI * 2);
      fgCtx.stroke();
      fgCtx.restore();
    }

    // Thruster Jet Plasma Sparks
    for (let i = thrusterParticles.length - 1; i >= 0; i--) {
      const p = thrusterParticles[i];
      p.x += p.vx;
      p.y += p.vy;
      p.size *= 0.94;
      p.alpha -= 0.035;

      if (p.alpha <= 0 || p.size < 0.5) {
        thrusterParticles.splice(i, 1);
        continue;
      }

      fgCtx.save();
      fgCtx.fillStyle = p.color;
      fgCtx.shadowColor = p.color;
      fgCtx.shadowBlur = 8;
      fgCtx.globalAlpha = p.alpha;
      fgCtx.beginPath();
      fgCtx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
      fgCtx.fill();
      fgCtx.restore();
    }

    // Speed Lines
    for (let i = speedLines.length - 1; i >= 0; i--) {
      const line = speedLines[i];
      line.x -= line.speed;
      if (line.x + line.length < 0) {
        speedLines.splice(i, 1);
        continue;
      }
      fgCtx.save();
      fgCtx.strokeStyle = '#00f3ff';
      fgCtx.globalAlpha = line.alpha;
      fgCtx.lineWidth = 1.5;
      fgCtx.beginPath();
      fgCtx.moveTo(line.x, line.y);
      fgCtx.lineTo(line.x + line.length, line.y);
      fgCtx.stroke();
      fgCtx.restore();
    }
  }

  // Update Loop
  let lastTime = performance.now();

  function animate(now) {
    const dt = Math.min(50, now - lastTime) / 1000;
    lastTime = now;

    const theme = themes[currentThemeIdx];

    // Smooth rider position following
    state.vx = (state.targetX - state.x) * 4.5;
    state.vy = (state.targetY - state.y) * 4.5;
    state.x += state.vx * dt;
    state.y += state.vy * dt;

    // Board tilt responds to lateral motion + steer
    const lateralSpeed = state.vx / width;
    state.targetTilt = lateralSpeed * 22;
    state.tilt += (state.targetTilt - state.tilt) * 6 * dt;

    // Handle Jump / Trick physics
    if (state.isJumping) {
      state.jumpTimer += dt * 2.8;
      const progress = state.jumpTimer;
      
      // Arc formula
      state.jumpOffsetY = -Math.sin(progress * Math.PI) * 140;

      if (state.trickType === 0) {
        // Ollie: pop nose up then level out
        state.jumpRotation = -Math.sin(progress * Math.PI * 2) * 18;
      } else if (state.trickType === 1) {
        // 360 Spin trick
        state.jumpRotation = progress * 360;
      } else {
        // Kickflip / Barrel roll
        state.jumpRotation = Math.sin(progress * Math.PI) * 45;
      }

      if (progress >= 1) {
        state.isJumping = false;
        state.jumpOffsetY = 0;
        state.jumpRotation = 0;
        // Landing dust ring
        roadParticles.push({
          x: state.x,
          y: state.y + 195,
          rx: 15,
          ry: 6,
          maxRadius: 60,
          alpha: 0.9,
          color: '#05ffa1'
        });
      }
    } else {
      state.jumpOffsetY = 0;
      state.jumpRotation = 0;
    }

    // Speed display interpolation
    state.targetSpeed = (state.turbo ? 142 : 88) + Math.sin(now * 0.005) * 4;
    state.speed += (state.targetSpeed - state.speed) * 4 * dt;
    speedDisplay.textContent = Math.round(state.speed);

    // Apply transforms to Capybara DOM container
    const totalRotation = state.tilt + state.jumpRotation;
    riderContainer.style.transform = `translate(${state.x - 230}px, ${state.y - 230 + state.jumpOffsetY}px) rotate(${totalRotation}deg) scale(${state.turbo ? 1.05 : 1})`;

    spawnThrusterParticles();
    drawBackground(theme);
    drawForeground();

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
})();
